local Settings = {file = "DeathCrushSettings.json", data = {}, extra = {}}
do
	local HttpService = game:GetService("HttpService")
	local okRead, raw = pcall(function()
		return typeof(isfile) == "function" and isfile(Settings.file) and readfile(Settings.file) or nil
	end)
	if okRead and type(raw) == "string" and raw ~= "" then
		local okJson, t = pcall(HttpService.JSONDecode, HttpService, raw)
		if okJson and type(t) == "table" then Settings.data = t end
	end
	local queued = false
	function Settings.save()
		if queued or typeof(writefile) ~= "function" then return end
		queued = true
		task.delay(0.5, function()
			queued = false
			local out = {toggles = {}, extra = {}}
			for k, v in pairs(state) do if type(v) == "boolean" then out.toggles[k] = v end end
			for k, get in pairs(Settings.extra) do
				local okGet, v = pcall(get)
				if okGet then out.extra[k] = v end
			end
			pcall(function() writefile(Settings.file, HttpService:JSONEncode(out)) end)
		end)
	end
	function Settings.value(name, get)
		Settings.extra[name] = get
		local e = Settings.data.extra
		return type(e) == "table" and e[name] or nil
	end
	local saved = Settings.data.toggles
	if type(saved) == "table" then
		for k, v in pairs(saved) do
			if state[k] ~= nil and type(v) == "boolean" then state[k] = v end
		end
	end
	state.killAura = false
end

local function make(class, props, children)
	local obj = Instance.new(class)
	for k, v in pairs(props or {}) do
		if k ~= "Parent" then obj[k] = v end
	end
	for _, child in ipairs(children or {}) do child.Parent = obj end
	if props and props.Parent then obj.Parent = props.Parent end
	return obj
end

		local p = c.Parent
		if not (p and p:IsA("GuiObject")) or c.CornerRadius.Scale ~= 0 then return end
		local s = p.Size
		local w = s.X.Scale == 0 and s.X.Offset or math.huge
		local h = s.Y.Scale == 0 and s.Y.Offset or math.huge
		local small = math.min(w, h)
		if small > 0 and c.CornerRadius.Offset * 2 >= small - 1 then return end
		c.CornerRadius = SHARP
	end

		for _, d in ipairs(root:GetDescendants()) do pcall(skinOne, d) end

local function rootOf(c) return c and c:FindFirstChild("HumanoidRootPart") end
local function humOf(c) return c and c:FindFirstChildOfClass("Humanoid") end
local function isAlive(c) local h = humOf(c); return h ~= nil and h.Health > 0 end
local function teamOf(plr) return plr.Team and plr.Team.Name or "" end
local function isKiller(plr) return teamOf(plr):lower():find("kill", 1, true) ~= nil end
local function isVictim(plr) return teamOf(plr):lower():find("vict", 1, true) ~= nil end
local function roundLive() return workspace:GetAttribute("Round") == true end
local function confValue(char, name)
	local conf = char and char:FindFirstChild("Conf")
	local v = conf and conf:FindFirstChild(name)
	return v and v.Value
end

local function nearestPlayer(maxDist, filter)
	local myRoot = rootOf(localPlayer.Character)
	if not myRoot then return nil end
	local best, bestDist = nil, maxDist
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= localPlayer and isAlive(plr.Character) and filter(plr) then
			local r = rootOf(plr.Character)
			local d = r and (r.Position - myRoot.Position).Magnitude
			if d and d < bestDist then best, bestDist = plr, d end
		end
	end
	return best, bestDist
end

local function teleportTo(cf)
	local root = rootOf(localPlayer.Character)
	if not root then return false end
	root.AssemblyLinearVelocity = Vector3.zero
	root.CFrame = cf
	return true
end

local generators = {}
local function genProgress(g)
	local p = g:FindFirstChild("Progress")
	local raw = (p and p:IsA("ValueBase") and tonumber(p.Value)) or 0
	return math.min(100, raw / 90 * 100)
end
local function genPart(g)
	return g.PrimaryPart or g:FindFirstChildWhichIsA("BasePart", true)
end
local function addGenerator(d)
	if d:IsA("Model") and d.Name == "Generator" then generators[d] = true end
end
local genScan = {alive = true, lastRound = nil}
onCleanup(function() genScan.alive = false end)
function genScan.run()
	for i, d in ipairs(workspace:GetDescendants()) do
		addGenerator(d)
		if i % 3000 == 0 then task.wait() end
	end
end
task.spawn(function()
	genScan.run()
	while genScan.alive do
		task.wait(5)
		local round = workspace:GetAttribute("Round")
		if round ~= genScan.lastRound then
			genScan.lastRound = round
			if round == true then
				task.wait(3)
				genScan.run()
			end
		end
	end
end)
track(workspace.DescendantAdded:Connect(addGenerator))
local function liveGenerators()
	local list = {}
	for g in pairs(generators) do
		if not (g.Parent and g:IsDescendantOf(workspace)) then
			generators[g] = nil
		elseif genPart(g) then
			table.insert(list, g)
		end
	end
	return list
end
local function gatesList()

	for _, g in ipairs(CollectionService:GetTagged("Gate")) do
		if g:IsA("BasePart") and g:IsDescendantOf(workspace) then table.insert(list, g) end
	end
	return list
end

local espTick = 0
track(RunService.Heartbeat:Connect(function()
	if os.clock() - espTick < 0.2 then return end
	espTick = os.clock()
	local myRoot = rootOf(localPlayer.Character)
	local function dist(part) return (part and myRoot) and math.floor((part.Position - myRoot.Position).Magnitude) or 0 end

	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= localPlayer then
			local char = plr.Character
			local e = espItems[plr]
			if char and (not e or e.model ~= char) and char:FindFirstChild("Head") then
				e = makeEsp(plr, char, char.Head)
			end
			if e then
				local killer, victim = isKiller(plr), isVictim(plr)
				local show = e.model.Parent ~= nil and isAlive(e.model)
					and ((killer and state.killerEsp) or (victim and state.victimEsp))
				local text = ""
				if show then
					local root = rootOf(e.model)
					if killer then
						text = string.format("%s\nKILLER  %dm", plr.DisplayName, dist(root))
					else
						local hp, max = confValue(e.model, "Health"), confValue(e.model, "MaxHealth")
						text = string.format("%s\n%s  %dm", plr.DisplayName,
							hp and string.format("%d/%d HP", hp, max or 0) or "Victim", dist(root))
					end
				end
				setEsp(e, show, killer and COLORS.killer or COLORS.victim, text)
			end
		end
	end

	for _, g in ipairs(liveGenerators()) do
		local e = espItems[g]
		if not e then e = makeEsp(g, g, genPart(g)) end
		local p = genProgress(g)
		setEsp(e, state.genEsp and p < 100, COLORS.generator, string.format("Generator %d%%\n%dm", math.floor(p), dist(genPart(g))))
	end
	for _, gate in ipairs(gatesList()) do
		local e = espItems[gate]
		if not e then e = makeEsp(gate, gate, gate) end
		setEsp(e, state.gateEsp, COLORS.gate, string.format("EXIT GATE\n%dm", dist(gate)))
	end
	for key, e in pairs(espItems) do
		if typeof(key) == "Instance" and not key:IsA("Player") and not key:IsDescendantOf(workspace) then removeEsp(key) end
	end
end))

local function genStand(part)
	-- stand on the back side of the generator; if a wall is there, fall back to the front
	local cf = part.CFrame
	local center = cf.Position + Vector3.new(0, 2, 0)
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = {part:FindFirstAncestorOfClass("Model") or part, localPlayer.Character}
	for _, dir in ipairs({-cf.LookVector, cf.LookVector, cf.RightVector, -cf.RightVector}) do
		local flat = Vector3.new(dir.X, 0, dir.Z)
		if flat.Magnitude > 0.1 then
			flat = flat.Unit
			if not workspace:Raycast(center, flat * 5, params) then
				local spot = center + flat * 4
				return CFrame.new(spot, Vector3.new(cf.Position.X, spot.Y, cf.Position.Z))
			end
		end
	end
	local spot = center - cf.LookVector * 4
	return CFrame.new(spot, Vector3.new(cf.Position.X, spot.Y, cf.Position.Z))
end

	teleportTo(genStand(part))

end

if typeof(hookmetamethod) == "function" and typeof(getnamecallmethod) == "function" then
	local isOurs = typeof(checkcaller) == "function" and checkcaller or function() return false end
	local wrap = typeof(newcclosure) == "function" and newcclosure or function(f) return f end
	local old
	old = hookmetamethod(game, "__namecall", wrap(function(self, ...)
		if self == TweenService and not Speed.dead and state.fastVault and not isOurs() and getnamecallmethod() == "Create" then
			local obj, info, props = ...
			local root = localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart")
			if obj == root and typeof(info) == "TweenInfo" and type(props) == "table" and props.CFrame then
				info = TweenInfo.new(info.Time / Speed.factor, info.EasingStyle, info.EasingDirection,
					info.RepeatCount, info.Reverses, info.DelayTime)
				task.defer(Speed.boostAnims)
				return old(self, obj, info, props)
			end
		end
		return old(self, ...)
	end))
else
	infoLabel(killerPage, "Your executor can't hook calls, fast vault won't work.", 30)
end

function Speed.isSlowdown(...)
	local args = {...}
	if typeof(args[1]) == "Instance" and args[1]:IsA("Player") then table.remove(args, 1) end
	local op, value = args[1], tonumber(args[2])
	if typeof(op) == "boolean" or not value then return false end
	if op == "=" then return value < 16 end
	if op == "-" then return value > 0 end
	if op == "*" then return value < 1 end
	if op == "/" then return value > 1 end
	return false
end

function Speed.hookSignal(signal)
	if typeof(getconnections) ~= "function" then return 0 end
	local ok, conns = pcall(getconnections, signal)
	if not ok then return 0 end
	local n = 0
	for _, c in ipairs(conns) do
		local fn = c.Function
		if fn and not Speed.ours[fn] and pcall(function() c:Disable() end) then
			n += 1
			local wrapper
			wrapper = function(...)
				if state.noSlow and not Speed.dead and Speed.isSlowdown(...) then return end
				return fn(...)
			end
			Speed.ours[wrapper] = true
			track(signal:Connect(wrapper))
			onCleanup(function() pcall(function() c:Enable() end) end)
		end
	end
	return n
end

function Speed.hookAll()
	local rs = ReplicatedStorage:FindFirstChild("RemoteEvents")
	local remoteEv = rs and rs:FindFirstChild("SpeedChanger")
	local bindables = rs and rs:FindFirstChild("Bindables")
	local localEv = bindables and bindables:FindFirstChild("SpeedChangerLocal")
	local n = 0
	if remoteEv then n += Speed.hookSignal(remoteEv.OnClientEvent) end
	if localEv then n += Speed.hookSignal(localEv.Event) end
	return n
end
if typeof(getconnections) ~= "function" then
	infoLabel(killerPage, "Your executor has no getconnections, no slowdowns won't work.", 30)
else
	task.spawn(Speed.hookAll)
	track(localPlayer.CharacterAdded:Connect(function()
		task.wait(3)
		Speed.hookAll()
	end))
end

track(RunService.Heartbeat:Connect(function()
	local hum = humOf(localPlayer.Character)
	if not hum then return end
	if hum ~= Move.hum then
		Move.hum = hum
		Move.gameSpeed, Move.appliedSpeed, Move.gameJump, Move.appliedJump = nil, nil, nil, nil
	end
	if state.speedOn and hum.WalkSpeed > 0 and hum.WalkSpeed ~= Move.speed then
		if hum.WalkSpeed ~= Move.appliedSpeed then Move.gameSpeed = hum.WalkSpeed end
		hum.WalkSpeed = Move.speed
		Move.appliedSpeed = Move.speed
	end
	if state.jumpOn then
		if not hum:GetStateEnabled(Enum.HumanoidStateType.Jumping) then
			pcall(hum.SetStateEnabled, hum, Enum.HumanoidStateType.Jumping, true)
		end
		local prop, target = Move.jumpProp(hum), Move.jumpTarget(hum)
		local cur = hum[prop]
		if math.abs(cur - target) > 0.01 then
			if not Move.appliedJump or math.abs(cur - Move.appliedJump) > 0.01 then Move.gameJump = cur end
			hum[prop] = target
			Move.appliedJump = target
		end
	end
end))

-- Admin detection. Runs on its own so it still protects you when the main hub menu isn't open.
-- If the hub is open, its own detector shows the alert and calls our panic, so this one stays quiet.
do
	local ROBLOX_STAFF_GROUP = 1200769
	local STAFF_WORDS = {"admin", "mod", "staff", "owner", "founder", "developer", "dev", "creator", "manager", "helper", "head"}
	local alerted = {}
	local holder = make("Frame", {BackgroundTransparency = 1, AnchorPoint = Vector2.new(0.5, 0),
		Position = UDim2.new(0.5, 0, 0, 20), Size = UDim2.new(0, 460, 1, -40), Parent = UI.toastGui,
	}, {make("UIListLayout", {Padding = UDim.new(0, 8), HorizontalAlignment = Enum.HorizontalAlignment.Center,
		SortOrder = Enum.SortOrder.LayoutOrder})})
	local count = 0

	local function reasonsFor(plr)
		local reasons, isOwner = {}, false
		if game.CreatorType == Enum.CreatorType.User and plr.UserId == game.CreatorId then
			isOwner = true
			table.insert(reasons, "Owns this game")
		elseif game.CreatorType == Enum.CreatorType.Group then
			local okR, rank = pcall(plr.GetRankInGroup, plr, game.CreatorId)
			local okN, role = pcall(plr.GetRoleInGroup, plr, game.CreatorId)
			rank = okR and rank or 0
			role = okN and role or "?"
			local staffWord = false
			for _, w in ipairs(STAFF_WORDS) do
				if role:lower():find(w, 1, true) then staffWord = true; break end
			end
			if rank == 255 then
				isOwner = true
				table.insert(reasons, "Owns this game (owner of the game's group: " .. role .. ")")
			elseif rank >= 254 then
				table.insert(reasons, "Top rank in the game's group: " .. role .. " (" .. rank .. ")")
			elseif rank > 0 and staffWord then
				table.insert(reasons, "Staff role in the game's group: " .. role .. " (" .. rank .. ")")
			elseif rank >= 200 then
				table.insert(reasons, "High rank in the game's group: " .. role .. " (" .. rank .. ")")
			end
		end
		local okS, inStaff = pcall(plr.IsInGroup, plr, ROBLOX_STAFF_GROUP)
		if okS and inStaff then table.insert(reasons, "Roblox staff member") end
		return reasons, isOwner
	end

	local function showAlert(plr, reasons, isOwner, wasHere, off)
		count += 1
		local card = make("Frame", {BackgroundColor3 = THEME.bg, BorderSizePixel = 0, Size = UDim2.new(1, 0, 0, 120),
			LayoutOrder = count, Parent = holder}, {corner(12)})
		local border = stroke(THEME.accent, 3, 0)
		border.Parent = card
		TweenService:Create(border, TweenInfo.new(0.6, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true),
			{Transparency = 0.7}):Play()
		local avatar = make("ImageLabel", {BackgroundColor3 = THEME.panel, BorderSizePixel = 0,
			Position = UDim2.new(0, 14, 0.5, -36), Size = UDim2.fromOffset(72, 72), Parent = card}, {corner(36)})
		task.spawn(function()
			local ok, img = pcall(Players.GetUserThumbnailAsync, Players, plr.UserId,
				Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size150x150)
			if ok and avatar.Parent then avatar.Image = img end
		end)
		make("TextLabel", {BackgroundTransparency = 1, Position = UDim2.fromOffset(100, 8), Size = UDim2.new(1, -150, 0, 26),
			Font = Enum.Font.GothamBlack, Text = isOwner and "GAME OWNER DETECTED" or "ADMIN DETECTED", TextSize = 20,
			TextColor3 = THEME.accent, TextXAlignment = Enum.TextXAlignment.Left, Parent = card})
		make("TextLabel", {BackgroundTransparency = 1, Position = UDim2.fromOffset(100, 34), Size = UDim2.new(1, -110, 0, 18),
			Font = Enum.Font.GothamBold, Text = plr.DisplayName .. "  (@" .. plr.Name .. ")", TextSize = 15,
			TextColor3 = THEME.text, TextXAlignment = Enum.TextXAlignment.Left, TextTruncate = Enum.TextTruncate.AtEnd, Parent = card})
		make("TextLabel", {BackgroundTransparency = 1, Position = UDim2.fromOffset(100, 54), Size = UDim2.new(1, -110, 0, 30),
			Font = Enum.Font.GothamMedium, Text = table.concat(reasons, "\n"), TextSize = 12,
			TextColor3 = Color3.fromRGB(255, 170, 170), TextWrapped = true, TextXAlignment = Enum.TextXAlignment.Left,
			TextYAlignment = Enum.TextYAlignment.Top, Parent = card})
		local status = make("TextLabel", {BackgroundTransparency = 1, Position = UDim2.new(0, 100, 1, -30),
			Size = UDim2.new(1, -110, 0, 26), Font = Enum.Font.Gotham, TextSize = 11, TextColor3 = THEME.subtext,
			TextWrapped = true, TextXAlignment = Enum.TextXAlignment.Left, TextYAlignment = Enum.TextYAlignment.Top, Parent = card,
			Text = (wasHere and "Already in the server" or "Joined") .. " at " .. os.date("%H:%M:%S") .. "  ·  "
				.. (#off > 0 and ("Turned off: " .. table.concat(off, ", ")) or "Nothing was on")})
		local close = make("TextButton", {BackgroundColor3 = THEME.panel, BorderSizePixel = 0, AutoButtonColor = false,
			AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, -8, 0, 8), Size = UDim2.fromOffset(28, 28),
			Font = Enum.Font.GothamBold, Text = "×", TextSize = 20, TextColor3 = THEME.text, Parent = card},
			{corner(6), stroke(THEME.accent, 1, 0.3)})
		track(close.Activated:Connect(function() UI.sfx("click"); card:Destroy() end))
		track(Players.PlayerRemoving:Connect(function(p)
			if p == plr and status.Parent then
				status.Text = status.Text .. "  ·  left at " .. os.date("%H:%M:%S")
				status.TextColor3 = THEME.good
			end
		end))
		task.spawn(function()
			for _ = 1, 3 do UI.sfx("error"); task.wait(0.25) end
		end)
	end

	local function check(plr, wasHere)
		if plr == localPlayer or alerted[plr] then return end
		local reasons, isOwner = reasonsFor(plr)
		if #reasons == 0 or not plr.Parent or not gui.Parent then return end
		alerted[plr] = true
		if type(_G.PlayerAnnoyerCleanup) == "function" then return end
		local okOff, off = pcall(turnAllOff)
		showAlert(plr, reasons, isOwner, wasHere, okOff and type(off) == "table" and off or {})
	end

	for _, p in ipairs(Players:GetPlayers()) do task.spawn(check, p, true) end
	track(Players.PlayerAdded:Connect(function(p) task.spawn(check, p, false) end))
end

_G.DeathCrushPanic = turnAllOff
_G.DeathCrushCleanup = function()
	for _, c in ipairs(connections) do pcall(function() c:Disconnect() end) end
	table.clear(connections)
	for _, fn in ipairs(cleanupTasks) do pcall(fn) end
	table.clear(cleanupTasks)
	if gui then gui:Destroy() end
	_G.DeathCrushCleanup = nil
	_G.DeathCrushPanic = nil
end
track(closeBtn.Activated:Connect(function() _G.DeathCrushCleanup() end))
