local Settings = {file = "FightClubSettings.json", data = {}, extra = {}}
do
	local HttpService = game:GetService("HttpService")
	local okRead, raw = pcall(function()
		return typeof(isfile) == "function" and isfile(Settings.file) and readfile(Settings.file) or nil
	end)
	if okRead and type(raw) == "string" and raw ~= "" then
		local okJson, t = pcall(HttpService.JSONDecode, HttpService, raw)
		if okJson and type(t) == "table" then Settings.data = t end
	end
	local queued = false
	function Settings.save()
		if queued or typeof(writefile) ~= "function" then return end
		queued = true
		task.delay(0.5, function()
			queued = false
			local out = {toggles = {}, extra = {}}
			for k, v in pairs(state) do if type(v) == "boolean" then out.toggles[k] = v end end
			for k, get in pairs(Settings.extra) do
				local okGet, v = pcall(get)
				if okGet then out.extra[k] = v end
			end
			pcall(function() writefile(Settings.file, HttpService:JSONEncode(out)) end)
		end)
	end
	function Settings.value(name, get)
		Settings.extra[name] = get
		local e = Settings.data.extra
		return type(e) == "table" and e[name] or nil
	end
	local saved = Settings.data.toggles
	if type(saved) == "table" then
		for k, v in pairs(saved) do
			if state[k] ~= nil and type(v) == "boolean" then state[k] = v end
		end
	end
end

local function make(class, props, children)
	local obj = Instance.new(class)
	for k, v in pairs(props or {}) do
		if k ~= "Parent" then obj[k] = v end
	end
	for _, child in ipairs(children or {}) do child.Parent = obj end
	if props and props.Parent then obj.Parent = props.Parent end
	return obj
end

local packetFolder = ReplicatedStorage:FindFirstChild("Packet")
local packetRemote = packetFolder and packetFolder:FindFirstChild("RemoteEvent")

local function sendPacket(name, writer)
	local id = packetRemote and packetRemote:GetAttribute(name)
	if type(id) ~= "number" then
		notify("Fight Club", "Packet '" .. name .. "' not found")
		return false
	end
	local parts = {string.char(id)}
	if writer then writer(parts) end
	packetRemote:FireServer(buffer.fromstring(table.concat(parts)))
	return true
end
local function writeString(parts, s) table.insert(parts, string.char(#s) .. s) end
local function writeAnyString(parts, s) table.insert(parts, "\11" .. string.char(#s) .. s) end
local function writeAnyNil(parts) table.insert(parts, "\0") end

local combatStats = {sent = 0, acked = 0, unanswered = 0, lastAck = 0, lastSent = 0}

local function fistPacket(cmd, arg)
	local ok = sendPacket("FistRemote", function(p)
		writeString(p, cmd)
		if arg then writeAnyString(p, arg) else writeAnyNil(p) end
		for _ = 1, 4 do writeAnyNil(p) end
	end)
	if ok then
		combatStats.sent += 1
		combatStats.unanswered += 1
		combatStats.lastSent = os.clock()
	end
	return ok
end
local function emotePacket(name)
	return sendPacket("Emote", function(p) writeString(p, name) end)
end

local function rootOf(c) return c and c:FindFirstChild("HumanoidRootPart") end
local function humOf(c) return c and c:FindFirstChildOfClass("Humanoid") end
local function isAlive(c) local h = humOf(c); return h ~= nil and h.Health > 0 end

local function fistsEquipped()
	local c = localPlayer.Character
	local tool = c and c:FindFirstChildOfClass("Tool")
	return tool ~= nil and tool.Name == "Fists"
end
local function canAttack()
	local c = localPlayer.Character
	return fistsEquipped() and isAlive(c) and localPlayer:GetAttribute("canAttack") ~= false
		and not c:GetAttribute("IsRagdolled") and not c:GetAttribute("Finish")
end

local function nearestEnemy(maxDist, from, includeSelf)
	local origin = from or rootOf(localPlayer.Character)
	if not origin then return nil end
	local best, bestDist
	for _, plr in ipairs(Players:GetPlayers()) do
		local c = plr.Character
		local r = rootOf(c)
		if (includeSelf or plr ~= localPlayer) and r and r ~= origin and isAlive(c) and not c:GetAttribute("IsRagdolled") then
			local d = (r.Position - origin.Position).Magnitude
			if d <= maxDist and (not bestDist or d < bestDist) then best, bestDist = plr, d end
		end
	end
	return best, bestDist
end

	for _, t in ipairs(UI.tabs) do
		local sel = t.name == name
		if sel and not t.page.Visible then
			t.page.Position = UDim2.new(0, 0, 0, 10)
			tween(t.page, {Position = UDim2.new(0, 0, 0, 0)}, TweenInfo.new(0.25, Enum.EasingStyle.Quint, Enum.EasingDirection.Out))
		end

		if sel then UI.crumb.Text = "›  " .. name .. (inGame and "" or "   <font color=\"#E41026\">wrong game</font>") end

-- One player picker style for every tab: a 2-column grid of buttons with the player's headshot.
-- opts.onClick(plr), opts.isSelected(plr) -> bool (highlights the button),
-- opts.nearest = {onClick = fn, isSelected = fn} adds a "Nearest player" button first.
local playerGrid
do
	local headshots, grids, alive = {}, {}, true
	onCleanup(function() alive = false end)
	local function paint(b, sel)
		local s = b:FindFirstChildOfClass("UIStroke")
		if s then s.Color = sel and THEME.accent or THEME.stroke end
		b.TextColor3 = sel and THEME.accent or THEME.text
	end
	local function avatarSlot(b)
		local pad = b:FindFirstChildOfClass("UIPadding")
		if pad then pad.PaddingLeft = UDim.new(0, 30) end
		b.TextXAlignment = Enum.TextXAlignment.Left
		return make("ImageLabel", {BackgroundColor3 = THEME.bg, BorderSizePixel = 0, AnchorPoint = Vector2.new(0, 0.5),
			Position = UDim2.new(0, -26, 0.5, 0), Size = UDim2.fromOffset(22, 22), Parent = b},
			{corner(11), stroke(THEME.stroke, 1, 0)})
	end
	local function labelOf(p) return p.DisplayName .. (isAlive(p.Character) and "" or "  (dead)") end

	local function refresh(g)
		local list = {}
		for _, p in ipairs(Players:GetPlayers()) do if p ~= localPlayer then table.insert(list, p) end end
		table.sort(list, function(a, b) return a.DisplayName:lower() < b.DisplayName:lower() end)
		local same = #list == #g.items
		if same then for i, p in ipairs(list) do if g.items[i] ~= p then same = false; break end end end
		if not same then
			for _, b in ipairs(g.btns) do b:Destroy() end
			table.clear(g.btns)
			g.items = list
			for i, p in ipairs(list) do
				local b = button(g.page, labelOf(p), function()
					g.opts.onClick(p)
					refresh(g)
				end, g.grid)
				b.LayoutOrder, b.Name = i, string.format("%03d", i)
				local img = avatarSlot(b)
				img.Image = headshots[p.UserId] or ""
				if not headshots[p.UserId] then
					task.spawn(function()
						local ok, url = pcall(Players.GetUserThumbnailAsync, Players, p.UserId,
							Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size48x48)
						if ok then headshots[p.UserId] = url; if img.Parent then img.Image = url end end
					end)
				end
				g.btns[i] = b
			end
			g.empty.Visible = #list == 0
		end
		for i, p in ipairs(g.items) do
			local b = g.btns[i]
			b.Text = labelOf(p)
			paint(b, g.opts.isSelected and g.opts.isSelected(p) or false)
		end
		if g.nearestBtn then paint(g.nearestBtn, g.opts.nearest.isSelected()) end
	end

	function playerGrid(page, opts)
		local g = {page = page, opts = opts, grid = buttonGrid(page), items = {}, btns = {}}
		if opts.nearest then
			g.nearestBtn = button(page, "Nearest player", function()
				opts.nearest.onClick()
				refresh(g)
			end, g.grid)
			g.nearestBtn.LayoutOrder, g.nearestBtn.Name = 0, "000"
			UI.icon("user", 16, avatarSlot(g.nearestBtn), THEME.text,
				{AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.5)})
		end
		g.empty = make("TextLabel", {BackgroundTransparency = 1, Name = "999", LayoutOrder = 999, Font = Enum.Font.Gotham,
			Text = "Nobody else in the server", TextSize = 11, TextColor3 = THEME.subtext,
			TextXAlignment = Enum.TextXAlignment.Left, Parent = g.grid})
		table.insert(grids, g)
		refresh(g)
		return {refresh = function() refresh(g) end}
	end
	task.spawn(function()
		while alive do
			task.wait(1)
			for _, g in ipairs(grids) do pcall(refresh, g) end
		end
	end)
end

local function logToFile(text)

	fileDirty = true
end
local function flushLog()
	if not fileDirty or typeof(writefile) ~= "function" then return end
	fileDirty = false
	pcall(writefile, LOG_FILE, table.concat(fileLines, "\n") .. "\n")
end

		task.wait(1)
	end
end)

	logToFile(text)
end

local function stripPlayerNames(s)
	for _, plr in ipairs(Players:GetPlayers()) do
		for _, n in ipairs({plr.Name, plr.DisplayName}) do
			if #n > 2 then
				local a, b = s:lower():find(n:lower(), 1, true)
				while a do
					s = s:sub(1, a - 1) .. s:sub(b + 1)
					a, b = s:lower():find(n:lower(), 1, true)
				end
			end
		end
	end
	return s
end

local function acWord(s)
	if type(s) ~= "string" or #s == 0 or #s > 4000 then return nil end
	local lower = stripPlayerNames(s):lower()
	for _, w in ipairs(AC_WORDS) do
		local hit = lower:match(w)
		if hit then return hit end
	end
	return nil
end

local function collectText(v, out, depth)
	depth = depth or 0
	if depth > 4 or #out > 200 then return end
	local t = typeof(v)
	if t == "string" then
		table.insert(out, v)
	elseif t == "buffer" then
		table.insert(out, buffer.tostring(v))
	elseif t == "Instance" then
		table.insert(out, v.Name)
	elseif t == "table" then
		for k, x in pairs(v) do
			collectText(k, out, depth + 1)
			collectText(x, out, depth + 1)
		end
	end
end

local function describe(...)
	local parts = {}
	for i = 1, select("#", ...) do
		local v = select(i, ...)
		local t = typeof(v)
		if t == "buffer" then
			local hex = buffer.tostring(v):sub(1, 200):gsub(".", function(c) return string.format("%02X ", c:byte()) end)
			table.insert(parts, "[" .. i .. "] buffer: " .. hex)
		elseif t == "table" then
			local texts = {}
			collectText(v, texts)
			table.insert(parts, "[" .. i .. "] table: " .. table.concat(texts, " | "):sub(1, 500))
		else
			table.insert(parts, "[" .. i .. "] " .. t .. ": " .. tostring(v):sub(1, 300))
		end
	end
	return table.concat(parts, "\n")
end

local guardStatus

local function trip(reason, detail, severe)
	log("[guard] " .. reason, THEME.accent)
	if detail and detail ~= "" then logToFile("    " .. detail:gsub("\n", "\n    ")) end
	guardStats.last = reason
	pcall(function()
		if typeof(writefile) == "function" then
			writefile("FightClubGuard_" .. os.date("%Y-%m-%d_%H-%M-%S") .. ".txt",
				"Place " .. game.PlaceId .. "  Job " .. game.JobId .. "\n" .. reason .. "\n\n" .. (detail or ""))
		end
	end)
	if not state.guard then return end
	local shouldKick = state.guardKick and (severe or state.paranoid)
	if not shouldKick then
		notify("Guard warning", reason)
		return
	end
	if tripped then return end
	tripped = true
	logToFile("KICKED: " .. reason)
	flushLog()
	state.autoBlock, state.autoPunch, state.autoKick, state.faceTarget = false, false, false, false
	localPlayer:Kick("Fight Club guard: " .. reason .. "\n(details saved to FightClubGuard_*.txt in the Xeno workspace)")
end

local adonisRemote
local function isAdonisRemote(r)
	return r.Parent == ReplicatedStorage and r.Name:match("^%x+%-%x+%-%x+%-%x+%-%x+$") ~= nil
		and r:FindFirstChild("__FUNCTION") ~= nil
end

local watched = {}
local function watchRemote(r)
	if watched[r] then return end
	watched[r] = true
	local full = r:GetFullName()
	local acName = AC_REMOTES[r.Name] or acWord(r.Name)
	if r:IsA("BindableEvent") then
		if not acName then return end
		track(r.Event:Connect(function(...)
			trip("anti-cheat bindable fired: " .. r.Name, describe(...), true)
		end))
		return
	end
	if isAdonisRemote(r) then adonisRemote = r end
	track(r.OnClientEvent:Connect(function(...)
		guardStats.checked += 1
		if acName then
			trip("anti-cheat remote fired: " .. r.Name, full .. "\n" .. describe(...), true)
			return
		end
		local texts = {}
		for i = 1, select("#", ...) do collectText(select(i, ...), texts) end
		for _, s in ipairs(texts) do
			local word = acWord(s)
			if word then
				trip(string.format("'%s' in %s", word, r.Name), full .. "\n" .. describe(...), true)
				return
			end
		end
		if r == adonisRemote then
			local cmd = select(3, ...)
			if type(cmd) == "string" and not ADONIS_OK[cmd] then
				trip("unusual Adonis command: " .. cmd, full .. "\n" .. describe(...), false)
			end
		end
	end))
end

local function maybeWatch(d)
	local c = d.ClassName
	if c == "RemoteEvent" or c == "UnreliableRemoteEvent" or c == "BindableEvent" then
		watchRemote(d)
	end
end

local function checkNewInstance(d)
	local c = d.ClassName
	if c == "LocalScript" or c == "ModuleScript" or c == "Script" or c == "RemoteEvent"
		or c == "RemoteFunction" or c == "BindableEvent" then
		local word = AC_REMOTES[d.Name] and d.Name or acWord(d.Name)
		if word then
			trip("anti-cheat object appeared: " .. d:GetFullName(), c, false)
		end
	end
end

local function guardRoot(root)
	pcall(function()
		for _, d in ipairs(root:GetDescendants()) do pcall(maybeWatch, d) end
	end)
	track(root.DescendantAdded:Connect(function(d)
		pcall(maybeWatch, d)
		pcall(checkNewInstance, d)
	end))
end

function Move.jumpProp(hum) return hum.UseJumpPower and "JumpPower" or "JumpHeight" end
function Move.jumpTarget(hum)
	return hum.UseJumpPower and math.sqrt(2 * workspace.Gravity * Move.jump) or Move.jump

function Move.restoreSpeed()
	local hum = Move.hum
	if hum and hum.Parent and Move.gameSpeed then hum.WalkSpeed = Move.gameSpeed end
	Move.gameSpeed, Move.appliedSpeed = nil, nil
end
function Move.restoreJump()
	local hum = Move.hum
	if hum and hum.Parent and Move.gameJump then hum[Move.jumpProp(hum)] = Move.gameJump end
	Move.gameJump, Move.appliedJump = nil, nil
end
onCleanup(function() Move.restoreSpeed(); Move.restoreJump() end)

local function distanceToModel(model, pos)
	local cf, size = model:GetBoundingBox()
	local p = cf:PointToObjectSpace(pos)
	local h = size / 2
	local clamped = Vector3.new(math.clamp(p.X, -h.X, h.X), math.clamp(p.Y, -h.Y, h.Y), math.clamp(p.Z, -h.Z, h.Z))
	return (p - clamped).Magnitude
end

local function inSafeZone()
	local folder = workspace:FindFirstChild("safezone")
	local zone = folder and folder:FindFirstChild("Safezone")
	local r = rootOf(localPlayer.Character)
	if not (zone and zone:IsA("BasePart") and r) then return false, zone end
	local p, h = zone.CFrame:PointToObjectSpace(r.Position), zone.Size / 2
	return math.abs(p.X) <= h.X and math.abs(p.Y) <= h.Y and math.abs(p.Z) <= h.Z, zone
end
local leaveSafeZone
do
	local leaving = false
	function leaveSafeZone()
		local inside, zone = inSafeZone()
		if not inside then return true end
		while leaving do task.wait(0.1) end
		local char = localPlayer.Character
		local r = rootOf(char)
		if not r or not inSafeZone() then return localPlayer:GetAttribute("canAttack") ~= false end
		leaving = true
		local p, h = zone.CFrame:PointToObjectSpace(r.Position), zone.Size / 2
		local dx, dz = h.X - math.abs(p.X), h.Z - math.abs(p.Z)
		local outLocal = dx <= dz and Vector3.new((p.X < 0 and -1 or 1) * (h.X + 5), p.Y, p.Z)
			or Vector3.new(p.X, p.Y, (p.Z < 0 and -1 or 1) * (h.Z + 5))
		local goal = zone.CFrame:PointToWorldSpace(outLocal)
		local params = RaycastParams.new()
		params.FilterType = Enum.RaycastFilterType.Exclude
		params.FilterDescendantsInstances = {char, zone.Parent}
		local hit = workspace:Raycast(goal + Vector3.new(0, 8, 0), Vector3.new(0, -60, 0), params)
		local y = hit and hit.Position.Y + 3 or r.Position.Y
		r.AssemblyLinearVelocity = Vector3.zero
		r.CFrame = CFrame.new(goal.X, y, goal.Z) * r.CFrame.Rotation
		local t0 = os.clock()
		task.wait(0.2)
		while localPlayer:GetAttribute("canAttack") == false and os.clock() - t0 < 3 and char.Parent do task.wait(0.1) end
		leaving = false
		local ok = localPlayer:GetAttribute("canAttack") ~= false
		log(string.format("[safezone] hopped out, canAttack=%s after %.1fs", tostring(localPlayer:GetAttribute("canAttack")),
			os.clock() - t0), ok and THEME.good or THEME.warn)
		return ok
	end
	button(combatPage, "TP out of safe zone", function()
		if not inSafeZone() then notify("Fight Club", "You're not in the safe zone"); return end
		local ok = leaveSafeZone()
		notify("Fight Club", ok and "Out of the safe zone, you can hit" or "Out, but the server still says canAttack = off",
			ok and THEME.good or THEME.red)
	end)
	toggle(combatPage, "Auto TP out after respawn", "autoLeaveSafe", "Hops you out of the spawn safe zone so you can hit")
	local function onSpawn(char)
		char:WaitForChild("HumanoidRootPart", 10)
		task.wait(0.5)

local function findOrbitTarget(name)
	if name == "" then return (nearestEnemy(500)) end

	for _, plr in ipairs(Players:GetPlayers()) do

local function orbitReroll()
	orbit.speed = rand(ORBIT.speedRange[1], ORBIT.speedRange[2]) * orbit.chaos
	orbit.dir = math.random(0, 1) == 0 and -1 or 1
	orbit.baseHeight = rand(ORBIT.verticalRange[1], ORBIT.verticalRange[2])
	orbit.nextReroll = rand(ORBIT.randomizeEvery[1], ORBIT.randomizeEvery[2]) / orbit.chaos
end

function stopOrbit(reason)
	orbiting = false
	if orbit.conn then orbit.conn:Disconnect(); orbit.conn = nil end
	local char = localPlayer.Character
	local hum = humOf(char)
	if hum then hum.PlatformStand = false end
	local cam = workspace.CurrentCamera
	if cam and hum then
		cam.CameraType = Enum.CameraType.Custom
		cam.CameraSubject = hum
	end
	refreshOrbit()
	if reason then
		orbitStatus.Text = reason
		orbitStatus.TextColor3 = THEME.warn
		notify("Orbit", reason)
	end
end

local function orbitStep(dt)
	if orbit.hold then return end
	local plr = orbit.player
	if not plr or not plr.Parent then stopOrbit("Target left the game"); return end
	local target = plr.Character
	local head = target and target:FindFirstChild("Head")
	if not head or not isAlive(target) then return end

	local char = localPlayer.Character
	local myHum, myRoot = humOf(char), rootOf(char)
	if not myRoot or not myHum or myHum.Health <= 0 then return end

	orbit.nextReroll -= dt
	if orbit.nextReroll <= 0 then orbitReroll() end
	orbit.pulse += dt
	if orbit.pulse >= ORBIT.pulseCooldown then
		orbit.pulse = 0
		orbit.radius = ORBIT.radius + rand(-ORBIT.radiusChaos, ORBIT.radiusChaos) * orbit.chaos
		orbit.height = orbit.baseHeight + rand(-ORBIT.verticalChaos, ORBIT.verticalChaos) * orbit.chaos
		if math.random() < 0.1 * orbit.chaos then orbit.dir = -orbit.dir end
	end
	orbit.angle += orbit.speed * dt * orbit.dir

	local headPos = head.Position
	local pos = headPos + Vector3.new(math.cos(orbit.angle) * orbit.radius, orbit.height, math.sin(orbit.angle) * orbit.radius)
	local cf = CFrame.lookAt(pos, headPos)
	if ORBIT.spinSelf then
		local tilt = 0.6 * math.min(orbit.chaos, 2)
		cf *= CFrame.Angles(rand(-tilt, tilt), rand(-math.pi, math.pi), rand(-tilt, tilt))
	end
	myHum.PlatformStand = true
	myRoot.CFrame = cf
	myRoot.AssemblyLinearVelocity = Vector3.zero
	myRoot.AssemblyAngularVelocity = Vector3.zero

	local cam = workspace.CurrentCamera
	local tHum = humOf(target)
	if cam and tHum and cam.CameraSubject ~= tHum then
		cam.CameraType = Enum.CameraType.Custom
		cam.CameraSubject = tHum
	end
end

local function startOrbit()
	local name = orbitPick
	local plr = findOrbitTarget(name)
	if not plr then
		orbitStatus.Text = name == "" and "Nobody to orbit" or ("@" .. name .. " isn't in this server - pick someone else")
		orbitStatus.TextColor3 = THEME.accent
		return
	end
	orbit.player = plr
	orbiting = true
	if inSafeZone() then
		orbit.hold = true
		orbitStatus.Text = "Leaving safe zone first..."
		leaveSafeZone()
		orbit.hold = false
		if not orbiting then return end
	end
	orbitReroll()
	orbit.conn = RunService.Heartbeat:Connect(orbitStep)
	refreshOrbit()
	log("orbiting " .. plr.DisplayName, THEME.good)
end

local lastBlock, lastPunch, lastKick, nextKickHigh = 0, 0, 0, true
local function onEnemyAnimation(plr, animTrack)
	if not state.autoBlock then return end
	local anim = animTrack.Animation
	local id = anim and tostring(anim.AnimationId):match("%d+")
	if not (id and attackAnims[id]) or not canAttack() then return end
	local myRoot, theirRoot = rootOf(localPlayer.Character), rootOf(plr.Character)
	if not (myRoot and theirRoot) then return end
	local offset = myRoot.Position - theirRoot.Position
	if offset.Magnitude > BLOCK_RANGE or offset.Magnitude < 0.01 then return end
	if theirRoot.CFrame.LookVector:Dot(offset.Unit) < 0.2 then return end
	if os.clock() - lastBlock < 0.35 then return end
	lastBlock = os.clock()
	fistPacket("keypress", "f")
	log("auto-blocked " .. plr.DisplayName .. "'s " .. attackAnims[id], THEME.good)
end

local function nameOf(plr) return plr == localPlayer and "You" or plr.DisplayName end

local function watchCharacter(plr, char)
	local conns = playerConns[plr]
	for _, c in ipairs(conns.char) do c:Disconnect() end
	table.clear(conns.char)

	local hum = char:WaitForChild("Humanoid", 10)
	if not hum or plr.Character ~= char then return end

	local lastHealth = hum.Health
	table.insert(conns.char, hum.HealthChanged:Connect(function(health)
		local dmg = lastHealth - health
		lastHealth = health
		if not state.hitLog or dmg < 0.5 then return end
		local attacker = nearestEnemy(12, rootOf(char), true)
		if attacker == plr then attacker = nil end
		local by = attacker and (nameOf(attacker) .. " hit ") or ""
		local text = string.format("%s%s  -%d  (%d hp)", by, nameOf(plr), math.floor(dmg + 0.5), math.max(0, math.floor(health)))
		local color = (plr == localPlayer and THEME.accent) or (attacker == localPlayer and THEME.good) or THEME.text
		log(text, color)
		if health <= 0 then log(nameOf(plr) .. " died" .. (attacker and (" (" .. nameOf(attacker) .. ")") or ""), color) end
	end))

	if plr == localPlayer then return end

	local animator = hum:WaitForChild("Animator", 10)
	if animator then
		table.insert(conns.char, animator.AnimationPlayed:Connect(function(animTrack)
			onEnemyAnimation(plr, animTrack)
		end))
	end
	char:WaitForChild("Head", 10)
	makeEsp(plr, char)
end

local function watchPlayer(plr)
	if playerConns[plr] then return end
	playerConns[plr] = {char = {}, player = {}}
	table.insert(playerConns[plr].player, plr.CharacterAdded:Connect(function(char)
		task.spawn(watchCharacter, plr, char)
	end))
	if plr.Character then task.spawn(watchCharacter, plr, plr.Character) end
end

local function unwatchPlayer(plr)
	local conns = playerConns[plr]
	if conns then
		for _, c in ipairs(conns.char) do c:Disconnect() end
		for _, c in ipairs(conns.player) do c:Disconnect() end
		playerConns[plr] = nil
	end
	clearEsp(plr)
end

for _, plr in ipairs(Players:GetPlayers()) do watchPlayer(plr) end

			Text = (wasHere and "Already in the server" or "Joined") .. " at " .. os.date("%H:%M:%S") .. "  ·  "
				.. (#off > 0 and ("Turned off: " .. table.concat(off, ", ")) or "Nothing was on")})
		local close = make("TextButton", {BackgroundColor3 = THEME.panel, BorderSizePixel = 0, AutoButtonColor = false,
			AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, -8, 0, 8), Size = UDim2.fromOffset(28, 28),
			Font = Enum.Font.GothamBold, Text = "×", TextSize = 20, TextColor3 = THEME.text, Parent = card},
			{corner(6), stroke(THEME.accent, 1, 0.3)})
		track(close.Activated:Connect(function() UI.sfx("click"); card:Destroy() end))
		track(Players.PlayerRemoving:Connect(function(p)
			if p == plr and status.Parent then
				status.Text = status.Text .. "  ·  left at " .. os.date("%H:%M:%S")
				status.TextColor3 = THEME.good
			end
		end))
		task.spawn(function()
			for _ = 1, 3 do UI.sfx("error"); task.wait(0.25) end
		end)
	end

	local function check(plr, wasHere)
		if plr == localPlayer or alerted[plr] then return end
		local reasons, isOwner = reasonsFor(plr)
		if #reasons == 0 or not plr.Parent or not gui.Parent then return end
		alerted[plr] = true
		if type(_G.PlayerAnnoyerCleanup) == "function" then return end
		local okOff, off = pcall(_G.FightClubPanic)
		showAlert(plr, reasons, isOwner, wasHere, okOff and type(off) == "table" and off or {})
	end

	for _, p in ipairs(Players:GetPlayers()) do task.spawn(check, p, true) end
	track(Players.PlayerAdded:Connect(function(p) task.spawn(check, p, false) end))
end

local tpPage = newTab("Teleport", 5, "user")
do
	local function tpBehind(p)
		local myRoot = rootOf(localPlayer.Character)
		local r = p and p.Parent and rootOf(p.Character)
		if not myRoot then notify("Fight Club", "You have no character right now", THEME.red); return end
		if not r then notify("Fight Club", (p and p.DisplayName or "?") .. " has no character right now", THEME.red); return end
		if inSafeZone() then
			leaveSafeZone()
			myRoot, r = rootOf(localPlayer.Character), p.Parent and rootOf(p.Character)
			if not (myRoot and r) then return end
		end
		myRoot.AssemblyLinearVelocity = Vector3.zero
		myRoot.CFrame = r.CFrame * CFrame.new(0, 0, 3)
		notify("Fight Club", "Teleported to " .. p.DisplayName, THEME.good)
	end
	local function alivePlayers()
		local list = {}
		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= localPlayer and isAlive(p.Character) and rootOf(p.Character) then table.insert(list, p) end
		end
		return list
	end

	header(tpPage, "QUICK")
	local qgrid = buttonGrid(tpPage)
	button(tpPage, "Nearest player", function()
		local myRoot = rootOf(localPlayer.Character)
		local best, bestDist
		for _, p in ipairs(alivePlayers()) do
			local d = myRoot and (rootOf(p.Character).Position - myRoot.Position).Magnitude
			if d and (not bestDist or d < bestDist) then best, bestDist = p, d end
		end
		if not best then notify("Fight Club", "Nobody alive to teleport to"); return end
		tpBehind(best)
	end, qgrid)
	button(tpPage, "Random player", function()
		local list = alivePlayers()
		if #list == 0 then notify("Fight Club", "Nobody alive to teleport to"); return end
		tpBehind(list[math.random(#list)])
	end, qgrid)
	infoLabel(tpPage, "This game has an anti-cheat, long teleports can get you flagged or kicked.", 30)
	header(tpPage, "PLAYERS")
	playerGrid(tpPage, {onClick = tpBehind})
end

_G.FightClubCleanup = function()
	for _, c in ipairs(connections) do pcall(function() c:Disconnect() end) end
	table.clear(connections)
	for _, fn in ipairs(cleanupTasks) do pcall(fn) end
	table.clear(cleanupTasks)
	if gui then gui:Destroy() end
	_G.FightClubCleanup = nil
end
track(closeBtn.Activated:Connect(function() _G.FightClubCleanup() end))
