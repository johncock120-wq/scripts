local Settings = {file = "BlindButDangerousSettings.json", data = {}, extra = {}}
do
	local HttpService = game:GetService("HttpService")
	local okRead, raw = pcall(function()
		return typeof(isfile) == "function" and isfile(Settings.file) and readfile(Settings.file) or nil
	end)
	if okRead and type(raw) == "string" and raw ~= "" then
		local okJson, t = pcall(HttpService.JSONDecode, HttpService, raw)
		if okJson and type(t) == "table" then Settings.data = t end
	end
	local queued = false
	function Settings.save()
		if queued or typeof(writefile) ~= "function" then return end
		queued = true
		task.delay(0.5, function()
			queued = false
			local out = {toggles = {}, extra = {}}
			for k, v in pairs(state) do if type(v) == "boolean" then out.toggles[k] = v end end
			for k, get in pairs(Settings.extra) do
				local okGet, v = pcall(get)
				if okGet then out.extra[k] = v end
			end
			pcall(function() writefile(Settings.file, HttpService:JSONEncode(out)) end)
		end)
	end
	function Settings.value(name, get)
		Settings.extra[name] = get
		local e = Settings.data.extra
		return type(e) == "table" and e[name] or nil
	end
	local saved = Settings.data.toggles
	if type(saved) == "table" then
		for k, v in pairs(saved) do
			if state[k] ~= nil and type(v) == "boolean" then state[k] = v end
		end
	end
end

local function make(class, props, children)
	local obj = Instance.new(class)
	for k, v in pairs(props or {}) do
		if k ~= "Parent" then obj[k] = v end
	end
	for _, child in ipairs(children or {}) do child.Parent = obj end
	if props and props.Parent then obj.Parent = props.Parent end
	return obj
end

local function rootOf(c) return c and c:FindFirstChild("HumanoidRootPart") end
local function humOf(c) return c and c:FindFirstChildOfClass("Humanoid") end
local function isAlive(c) local h = humOf(c); return h ~= nil and h.Health > 0 end

local function roundLive() return ReplicatedStorage:GetAttribute("Etat") == "EnJeu" end
local function mode() return ReplicatedStorage:GetAttribute("ModeManche") end
local function inRound(plr) return plr:GetAttribute("EnManche") == true end

local function isSeeker(plr)
	local id = plr.UserId
	if id == (ReplicatedStorage:GetAttribute("AveugleUserId") or 0) then return true end
	if id == (ReplicatedStorage:GetAttribute("AveugleUserId2") or 0) then return true end
	return (mode() == "Anarchy" or plr:GetAttribute("Infecte") == true) and inRound(plr)
end
local function amSeeker() return roundLive() and isSeeker(localPlayer) end

local function findWeapon(plr)
	for _, holder in ipairs({plr.Character, plr:FindFirstChildOfClass("Backpack")}) do
		if holder then
			for _, t in ipairs(holder:GetChildren()) do
				if t:IsA("Tool") then
					if t:GetAttribute("ArmeMurder") then return "Murder bat" end
					if t.Name == "FusilAPompe" then return "Shotgun" end
				end
			end
		end
	end
	return nil
end

local function roleOf(plr)
	if not roundLive() or not inRound(plr) then return "lobby", "Lobby" end
	if plr:GetAttribute("Infecte") == true then return "infected", "Infected" end
	if isSeeker(plr) then return "seeker", (mode() == "Anarchy" and "Anarchy" or "BLIND MAN") end
	local w = findWeapon(plr)
	if w then return "armed", w end
	return "hider", "Hider"
end

local function nearest(maxDist, filter)
	local myRoot = rootOf(localPlayer.Character)
	if not myRoot then return nil end
	local best, bestDist = nil, maxDist
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= localPlayer and isAlive(plr.Character) and filter(plr) then
			local r = rootOf(plr.Character)
			local d = r and (r.Position - myRoot.Position).Magnitude
			if d and d < bestDist then best, bestDist = plr, d end
		end
	end
	return best, bestDist
end
local function isHiderTarget(plr) return inRound(plr) and not isSeeker(plr) end

local function faceTowards(pos)
	local root = rootOf(localPlayer.Character)
	if not root then return end
	local flat = Vector3.new(pos.X, root.Position.Y, pos.Z)
	if (flat - root.Position).Magnitude < 0.2 then return end
	root.CFrame = CFrame.lookAt(root.Position, flat)
end

local function nearestToCursor(filter)
	local cam = workspace.CurrentCamera
	local mousePos = UserInputService:GetMouseLocation()
	local best, bestDist = nil, 260
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= localPlayer and isAlive(plr.Character) and filter(plr) then
			local head = plr.Character:FindFirstChild("Head") or rootOf(plr.Character)
			if head then
				local sp, onScreen = cam:WorldToViewportPoint(head.Position)
				if onScreen then
					local d = (Vector2.new(sp.X, sp.Y) - mousePos).Magnitude
					if d < bestDist then best, bestDist = plr, d end
				end
			end
		end
	end
	return best
end

local changedHighlights = {}
local function isPlayerChar(m)
	return m and m:IsA("Model") and Players:GetPlayerFromCharacter(m) ~= nil and m ~= localPlayer.Character
end
local function gameHighlights()
	local list = {}
	local cam = workspace.CurrentCamera
	for _, child in ipairs(cam:GetChildren()) do
		if child.Name ~= "ContoursAveugle" then
			if child:IsA("Highlight") then table.insert(list, child) end
			for _, d in ipairs(child:GetDescendants()) do
				if d:IsA("Highlight") then table.insert(list, d) end
			end
		end
	end
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= localPlayer and plr.Character then
			for _, d in ipairs(plr.Character:GetChildren()) do
				if d:IsA("Highlight") then table.insert(list, d) end
			end
		end
	end
	return list
end
local function fixGameHighlights()
	if not state.esp then
		for h, mode in pairs(changedHighlights) do pcall(function() h.DepthMode = mode end) end
		table.clear(changedHighlights)
		return
	end
	for _, h in ipairs(gameHighlights()) do
		local target = h.Adornee or h.Parent
		if h.Enabled and isPlayerChar(target) and h.DepthMode ~= Enum.HighlightDepthMode.AlwaysOnTop then
			if changedHighlights[h] == nil then changedHighlights[h] = h.DepthMode end
			h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
		end

			local r = rootOf(plr.Character)
			if r and (r.Position - myRoot.Position).Magnitude < GRAB_RANGE + BLIND_SAFE_GAP then return true end
		end
	end
	return false
end

local VIM = (function()
	local ok, v = pcall(function() return game:GetService("VirtualInputManager") end)
	return ok and v or nil
end)()
local function hostageActive()
	local pg = localPlayer:FindFirstChildOfClass("PlayerGui")
	local otage = pg and pg:FindFirstChild("Otage")
	if otage and otage:IsA("ScreenGui") and otage.Enabled then return true end
	local char = localPlayer.Character
	return char ~= nil and char:GetAttribute("Porte") == true
end
local function tapKey(key)
	VIM:SendKeyEvent(true, key, false, game)
	VIM:SendKeyEvent(false, key, false, game)
end
local function expectedKey()
	local pg = localPlayer:FindFirstChildOfClass("PlayerGui")
	local otage = pg and pg:FindFirstChild("Otage")
	if otage then
		for _, l in ipairs(otage:GetChildren()) do
			if l:IsA("TextLabel") and (l.Text == "Q" or l.Text == "E") then
				return l.Text == "Q" and Enum.KeyCode.Q or Enum.KeyCode.E
			end
		end
	end
	return Enum.KeyCode.Q
end
local function qteHandlers()
	local out = {}
	if typeof(getconnections) ~= "function" then return out end
	local ok, conns = pcall(getconnections, UserInputService.InputBegan)
	if not ok or type(conns) ~= "table" then return out end
	for _, c in ipairs(conns) do
		local fn = c.Function
		if type(fn) == "function" then
			local okSrc, src = pcall(debug.info, fn, "s")
			if okSrc and type(src) == "string" and src:find("ControlesJoueur") then table.insert(out, fn) end
		end
	end
	return out
end
local function fakePresses(handlers, count)
	local key = expectedKey()
	local fired = false
	for _ = 1, count do
		local input = {KeyCode = key, UserInputType = Enum.UserInputType.Keyboard, UserInputState = Enum.UserInputState.Begin}
		for _, fn in ipairs(handlers) do
			if pcall(fn, input, false) then fired = true end
		end
		key = key == Enum.KeyCode.Q and Enum.KeyCode.E or Enum.KeyCode.Q
	end
	return fired
end

local escapeRunning = false
track(RunService.Heartbeat:Connect(function()
	if not state.autoEscape or escapeRunning or not hostageActive() then return end
	escapeRunning = true
	task.spawn(function()
		-- ControlesJoueur only sends Porter("debat") once per 0.1 s (ENVOI_DEBAT), but its bar fills on every
		-- press. Pressing faster than that fills the bar (which then sends "liberer" and locks full) before
		-- the server has counted enough debats, so the server refuses and the game retries every 2 s.
		-- One press per 0.1 s = every press reaches the server.
		local ESCAPE_PRESS_GAP = 0.105
		local useKeys = VIM ~= nil
		local presses, lastLiberer = 0, 0
		task.wait(0.1)
		while state.autoEscape and hostageActive() do
			local key = expectedKey()
			local handlers = qteHandlers()
			if #handlers > 0 and fakePresses(handlers, 1) then
			elseif useKeys and pcall(tapKey, key) then
			else
				useKeys = false
				if porter then pcall(function() porter:FireServer("debat") end) end
			end
			presses += 1
			if porter and presses >= 30 and os.clock() - lastLiberer > 0.5 then
				lastLiberer = os.clock()
				pcall(function() porter:FireServer("liberer") end)
			end
			task.wait(ESCAPE_PRESS_GAP)
		end
		escapeRunning = false
	end)
end))

track(RunService.Heartbeat:Connect(function()
	if not state.autoPush then pushStatus.Text = "Push: off"; return end
	if pushing then return end
	if not pousserRemote then pushStatus.Text = "Push: Pousser remote not found"; return end
	if not isAlive(localPlayer.Character) then pushStatus.Text = "Push: you're dead"; return end
	if localPlayer:GetAttribute("Passif") == "Pacifist" then pushStatus.Text = "Push: Pacifist passive can't push"; return end
	local cd = pushCooldown()
	local left = math.max(cd - (os.clock() - lastPush), 0)
	local readyAt = localPlayer:GetAttribute("PretA_Pousser")
	if type(readyAt) == "number" then left = math.max(left, readyAt - os.clock()) end
	if pushMode == 2 and not pushTarget then pushStatus.Text = "Push: no player selected"; return end

	local myRoot = rootOf(localPlayer.Character)
	local s, gap = nil, math.huge
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= localPlayer and isAlive(plr.Character) and pushFilter(plr) then
			local r = rootOf(plr.Character)
			if r and myRoot then
				local g = math.max((r.Position - myRoot.Position).Magnitude - 2, 0)
				if g < gap then s, gap = plr, g end
			end
		end
	end
	if not s then pushStatus.Text = "Push: no valid target (" .. PUSH_MODES[pushMode] .. ")"; return end
	if left > 0 then
		pushStatus.Text = string.format("Push: cooldown %.1fs   %s is %.1f studs away", left, s.DisplayName, gap)
		return
	end
	if gap > PUSH_RANGE then
		pushStatus.Text = string.format("Push: ready   %s is %.1f studs away (range %.1f)", s.DisplayName, gap, PUSH_RANGE)
		return
	end
	pushStatus.Text = "Push: pushing " .. s.DisplayName

	pushing = true
	lastPush = os.clock()
	task.spawn(function()
		local function stickTo()
			local r, me = rootOf(s.Character), rootOf(localPlayer.Character)
			if not (r and me) then return false end
			local hum = humOf(localPlayer.Character)
			local st = hum and hum:GetState()
			if st == Enum.HumanoidStateType.Physics or st == Enum.HumanoidStateType.Ragdoll
				or st == Enum.HumanoidStateType.FallingDown or (hum and hum.PlatformStand) then
				return false
			end
			-- only turn to face them, never move: teleporting next to a moving target could put us
			-- behind / inside them, which ragdolled us and dragged us along with their character
			local flat = Vector3.new(r.Position.X, me.Position.Y, r.Position.Z)
			if (flat - me.Position).Magnitude > 0.2 then
				me.CFrame = CFrame.lookAt(me.Position, flat)
			end
			return true
		end
		local t0 = os.clock()
		while os.clock() - t0 < 0.12 do
			if not stickTo() then break end
			RunService.Heartbeat:Wait()
		end
		local r = rootOf(s.Character)
		if r and isAlive(s.Character) and stickTo() then
			localPlayer:SetAttribute("PretA_Pousser", os.clock() + cd)
			local local_ = remote("CoupBatteLocal")
			if local_ then pcall(function() local_:Fire(localPlayer.Character, "pousse") end) end
			pcall(function() pousserRemote:FireServer() end)
		else
			lastPush = 0
		end
		pushing = false
	end)
end))

local lancer = remote("LancerBatteMurder")
local tirFusil = remote("TirFusil")
local function aimDir(plr)
	local head = localPlayer.Character and localPlayer.Character:FindFirstChild("Head")
	local tHead = plr.Character and (plr.Character:FindFirstChild("Head") or rootOf(plr.Character))
	if not (head and tHead) then return nil end
	local v = tHead.Position - head.Position
	return v.Magnitude > 0.1 and v.Unit or nil
end
local function otherAlive(p) return not roundLive() or inRound(p) end
local viseeFusil = remote("ViseeFusil")

local function pickAimTarget()
	return nearestToCursor(otherAlive) or nearest(80, otherAlive)
end

local function turnToward(t, seconds, sendPitch)
	local t0 = os.clock()
	local lastPitch = 0
	while os.clock() - t0 < seconds do
		local dir = aimDir(t)
		local r = rootOf(t.Character)
		if not (dir and r) then return false end
		faceTowards(r.Position)
		if sendPitch and viseeFusil and os.clock() - lastPitch > 0.05 then
			lastPitch = os.clock()
			pcall(function() viseeFusil:FireServer(math.asin(math.clamp(dir.Y, -1, 1))) end)
		end
		RunService.Heartbeat:Wait()
	end
	return true
end

local aimBusy = false
local function shotgunAt()
	local gun = localPlayer.Character and localPlayer.Character:FindFirstChild("FusilAPompe")
	if not gun then notify("Blind but Dangerous", "Equip the shotgun first"); return end
	if (localPlayer:GetAttribute("Cartouches") or 0) <= 0 and gun:GetAttribute("OwnerGun") ~= true then
		notify("Blind but Dangerous", "No shells"); return
	end
	local t = pickAimTarget()
	if not t then notify("Blind but Dangerous", "No target in view or within 80 studs"); return end
	if not turnToward(t, 0.12, true) then return end
	local dir = aimDir(t)
	if not dir then return end
	local cam = workspace.CurrentCamera
	local before = cam.CFrame
	cam.CFrame = CFrame.lookAt(cam.CFrame.Position, cam.CFrame.Position + dir)
	pcall(function() gun:Activate() end)
	task.defer(function()
		if (localPlayer:GetAttribute("PretA_FusilAPompe") or 0) < os.clock() - 0.05 then
			pcall(function() tirFusil:FireServer(dir, nil) end)
		end
	end)
	task.delay(0.05, function() pcall(function() cam.CFrame = before end) end)
end

local function throwBatAt()
	local bat = localPlayer.Character and localPlayer.Character:FindFirstChild("Batte")
	if not (bat and bat:GetAttribute("ArmeMurder")) then notify("Blind but Dangerous", "Equip the murder bat first"); return end
	if (bat:GetAttribute("BatteLanceeJusqua") or 0) > workspace:GetServerTimeNow() then return end
	local t = pickAimTarget()
	if not t then notify("Blind but Dangerous", "No target in view or within 80 studs"); return end
	if not turnToward(t, 0.12, false) then return end
	local dir = aimDir(t)
	if dir then pcall(function() lancer:FireServer(dir) end) end
end

local Move = {speed = 24, jump = 15, hum = nil, gameSpeed = nil, appliedSpeed = nil, gameJump = nil, appliedJump = nil}

function Move.jumpProp(hum) return hum.UseJumpPower and "JumpPower" or "JumpHeight" end
function Move.jumpTarget(hum)
	return hum.UseJumpPower and math.sqrt(2 * workspace.Gravity * Move.jump) or Move.jump
end
function Move.restoreSpeed()
	local hum = Move.hum
	if hum and hum.Parent and Move.gameSpeed then hum.WalkSpeed = Move.gameSpeed end
	Move.gameSpeed, Move.appliedSpeed = nil, nil

	if hum and hum.Parent and Move.gameJump then hum[Move.jumpProp(hum)] = Move.gameJump end
	Move.gameJump, Move.appliedJump = nil, nil
end
onCleanup(function() Move.restoreSpeed(); Move.restoreJump() end)

-- Stun effects (all client-side, driven by attributes the server puts on our Player):
--   Commotion script: concussion (CommotionT -> tunnel gui "CommotionTunnel", blur "CommotionFlou", colour
--     "CommotionCouleur", ringing "CommotionSifflement"/"CommotionSourd", muffled audio "CommotionEtouffe",
--     camera pulled around by render steps "CommotionAvant" + "Commotion") and flashbang (FlashT -> "FlashEcran" gui).
--   EffetPetard: firecracker nausea while EtourdiJusqua > now (camera wobble, blur, green tint, all sounds at 7%).
--     As Blind Man that same attribute also hides your noise outlines (VisionAveugle).
--   FauxBruits: Scary Serum fake footsteps / noises while FauxBruitsJusqua > now.
-- Setting the "until" attributes to 0 on our side makes the game end those effects itself (sounds restored).
local stunConns, stunDisabled = {}, {}
local STUN_UNTIL = {"EtourdiJusqua", "FauxBruitsJusqua"}
local function clearStunUntil(attr)
	local v = localPlayer:GetAttribute(attr)
	if type(v) == "number" and v > 0 then pcall(function() localPlayer:SetAttribute(attr, 0) end) end
end
local function killConcussion()
	local cam = workspace.CurrentCamera
	local flou = cam and cam:FindFirstChild("CommotionFlou")
	local pg = localPlayer:FindFirstChildOfClass("PlayerGui")
	local tunnel = pg and pg:FindFirstChild("CommotionTunnel")
	if not flou and not (tunnel and tunnel.Enabled) then return false end
	pcall(function() RunService:UnbindFromRenderStep("CommotionAvant") end)
	pcall(function() RunService:UnbindFromRenderStep("Commotion") end)
	if tunnel then tunnel.Enabled = false end
	for _, n in ipairs({"CommotionFlou", "CommotionCouleur"}) do
		local e = cam and cam:FindFirstChild(n)
		if e then e:Destroy() end
	end
	local SoundService = game:GetService("SoundService")
	for _, d in ipairs(SoundService:GetDescendants()) do
		if d.Name == "CommotionSifflement" or d.Name == "CommotionSourd" or d.Name == "CommotionEtouffe" then d:Destroy() end
	end
	return true
end
local stunFov = nil
local function setNoStun(on)
	for _, c in ipairs(stunConns) do c:Disconnect() end
	table.clear(stunConns)
	for _, c in ipairs(stunDisabled) do pcall(function() c:Enable() end) end
	table.clear(stunDisabled)
	if not on then return end

	-- best way: switch off the game's CommotionT / FlashT listeners so the effects never start
	if typeof(getconnections) == "function" then
		for _, attr in ipairs({"CommotionT", "FlashT"}) do
			local ok, conns = pcall(getconnections, localPlayer:GetAttributeChangedSignal(attr))
			if ok and type(conns) == "table" then
				for _, c in ipairs(conns) do
					local fn = c.Function
					local okSrc, src = pcall(debug.info, fn, "s")
					if type(fn) == "function" and okSrc and type(src) == "string" and src:find("Commotion") then
						if pcall(function() c:Disable() end) then table.insert(stunDisabled, c) end
					end
				end
			end
		end
	end
	for _, attr in ipairs(STUN_UNTIL) do
		clearStunUntil(attr)
		table.insert(stunConns, localPlayer:GetAttributeChangedSignal(attr):Connect(function() clearStunUntil(attr) end))
	end
	-- fallback (no getconnections): undo whatever started, every frame
	table.insert(stunConns, RunService.RenderStepped:Connect(function()
		local cam = workspace.CurrentCamera
		local pg = localPlayer:FindFirstChildOfClass("PlayerGui")
		local flash = pg and pg:FindFirstChild("FlashEcran")
		if flash and flash.Enabled then flash.Enabled = false end
		if killConcussion() then
			if cam and stunFov then cam.FieldOfView = stunFov end
		elseif cam then
			stunFov = cam.FieldOfView
		end
	end))
end
toggle(miscPage, "No stun effects", "noStun", "Concussion, flashbang white-out, firecracker nausea, Scary Serum fake noises", setNoStun)
onCleanup(function() setNoStun(false) end)

-- Shotgun recoil (ClientFusil recul): FOV +30 punch, then render step "ReculFusil<clock>" (priority Camera+1)
-- backflips the camera over 0.32 s (pitch 200 deg + roll 180 deg) so you end up facing BEHIND you.
-- Every shot sets the player attribute TirViewmodel, so on that we remember where the camera was looking and,
-- from a step that runs after the game's (Camera+2), put it back for the flip window and kill the FOV punch.
-- The knockback launch is movement, not camera, so it's left alone.
local RECOIL_STEP = "BBD_NoShotgunFlip"
local recoilConn, recoilUntil, recoilRot, recoilFov = nil, 0, nil, nil
local lastCamRot, lastCamFov = nil, nil
local FLIP_DOT = math.cos(math.rad(8))
local function setNoRecoil(on)
	if recoilConn then recoilConn:Disconnect(); recoilConn = nil end
	pcall(function() RunService:UnbindFromRenderStep(RECOIL_STEP) end)
	if not on then return end
	recoilConn = localPlayer:GetAttributeChangedSignal("TirViewmodel"):Connect(function()
		local cam = workspace.CurrentCamera
		if not cam then return end
		recoilRot = lastCamRot or cam.CFrame.Rotation
		recoilFov = localPlayer:GetAttribute("FOVBase") or lastCamFov or cam.FieldOfView
		recoilUntil = os.clock() + 0.45
		-- a new tween on the same property cancels the game's +30 FOV ease-out
		pcall(function() TweenService:Create(cam, TweenInfo.new(0), {FieldOfView = recoilFov}):Play() end)
	end)
	RunService:BindToRenderStep(RECOIL_STEP, Enum.RenderPriority.Camera.Value + 2, function()
		local cam = workspace.CurrentCamera
		if not cam then return end
		if os.clock() < recoilUntil and recoilRot then
			if cam.CameraType == Enum.CameraType.Custom and cam.CFrame.LookVector:Dot(recoilRot.LookVector) < FLIP_DOT then
				cam.CFrame = CFrame.new(cam.CFrame.Position) * recoilRot
			end
			if recoilFov then cam.FieldOfView = recoilFov end
		else
			lastCamRot, lastCamFov = cam.CFrame.Rotation, cam.FieldOfView
		end
	end)
end
toggle(miscPage, "No shotgun recoil flip", "noRecoil", "Your camera stays put when you fire (no backflip, no FOV punch). Knockback still happens", setNoRecoil)
onCleanup(function() setNoRecoil(false) end)

-- Forced first person: AnimObjetsHiders locks CameraMode = LockFirstPerson while you hold a tool, and for hiders in
-- Blackout rounds, unless the player attribute Screen3P is set (the devs' own third-person switch in DevUI, which
-- also sets Classic + zoom >= 20). As Blind Man it's VisionAveugle instead; "Full vision" already undoes that.
-- Holding a hostage also puts you in first person, but no client script does that, so it comes from the server
-- (or a hidden tool). To cover every source we also undo the lock each frame, before the camera updates.
local thirdConn
local THIRD_STEP = "BBD_NoFirstPerson"
local function setThirdPerson(on)
	if thirdConn then thirdConn:Disconnect(); thirdConn = nil end
	pcall(function() RunService:UnbindFromRenderStep(THIRD_STEP) end)
	pcall(function() localPlayer:SetAttribute("Screen3P", on or nil) end)
	if not on then return end
	RunService:BindToRenderStep(THIRD_STEP, Enum.RenderPriority.Camera.Value - 1, function()
		-- leave the Blind Man's blindfold view alone unless Full vision is on
		if amSeeker() and not state.fullVision then return end
		if localPlayer:GetAttribute("FOVBase") ~= nil and not state.fullVision then return end
		if localPlayer.CameraMode == Enum.CameraMode.LockFirstPerson then localPlayer.CameraMode = Enum.CameraMode.Classic end
		if localPlayer.CameraMaxZoomDistance < 20 then localPlayer.CameraMaxZoomDistance = 20 end
		if localPlayer.CameraMinZoomDistance > 0.5 and localPlayer.CameraMinZoomDistance >= localPlayer.CameraMaxZoomDistance then
			localPlayer.CameraMinZoomDistance = 0.5
		end
	end)
	thirdConn = localPlayer:GetAttributeChangedSignal("Screen3P"):Connect(function()
		if not localPlayer:GetAttribute("Screen3P") then pcall(function() localPlayer:SetAttribute("Screen3P", true) end) end
	end)
	task.defer(function()
		pcall(function()
			if localPlayer.CameraMode == Enum.CameraMode.LockFirstPerson and localPlayer:GetAttribute("FOVBase") == nil then
				localPlayer.CameraMode = Enum.CameraMode.Classic
			end
			localPlayer.CameraMaxZoomDistance = math.max(localPlayer.CameraMaxZoomDistance, 20)
		end)
	end)
end
toggle(miscPage, "No forced first person", "thirdPerson", "Stay in third person when holding items and in Blackout rounds", setThirdPerson)
onCleanup(function() if state.thirdPerson then setThirdPerson(false) end end)

	for key, set in pairs(toggleSetters) do
		if state[key] and key ~= "whistleWarn" and key ~= "dangerMeter" then
			pcall(set, false)
			table.insert(off, key)
		end
	end
	return off
end

-- Admin detection. Runs on its own so it still protects you when the main hub menu isn't open.
-- If the hub is open, its own detector shows the alert and calls our panic, so this one stays quiet.
do
	local ROBLOX_STAFF_GROUP = 1200769
	local STAFF_WORDS = {"admin", "mod", "staff", "owner", "founder", "developer", "dev", "creator", "manager", "helper", "head"}
	local alerted = {}
	local holder = make("Frame", {BackgroundTransparency = 1, AnchorPoint = Vector2.new(0.5, 0),
		Position = UDim2.new(0.5, 0, 0, 20), Size = UDim2.new(0, 460, 1, -40), Parent = UI.toastGui,
	}, {make("UIListLayout", {Padding = UDim.new(0, 8), HorizontalAlignment = Enum.HorizontalAlignment.Center,
		SortOrder = Enum.SortOrder.LayoutOrder})})
	local count = 0

	local function reasonsFor(plr)
		local reasons, isOwner = {}, false
		if game.CreatorType == Enum.CreatorType.User and plr.UserId == game.CreatorId then
			isOwner = true
			table.insert(reasons, "Owns this game")
		elseif game.CreatorType == Enum.CreatorType.Group then
			local okR, rank = pcall(plr.GetRankInGroup, plr, game.CreatorId)
			local okN, role = pcall(plr.GetRoleInGroup, plr, game.CreatorId)
			rank = okR and rank or 0
			role = okN and role or "?"
			local staffWord = false
			for _, w in ipairs(STAFF_WORDS) do
				if role:lower():find(w, 1, true) then staffWord = true; break end
			end
			if rank == 255 then
				isOwner = true
				table.insert(reasons, "Owns this game (owner of the game's group: " .. role .. ")")
			elseif rank >= 254 then
				table.insert(reasons, "Top rank in the game's group: " .. role .. " (" .. rank .. ")")
			elseif rank > 0 and staffWord then
				table.insert(reasons, "Staff role in the game's group: " .. role .. " (" .. rank .. ")")
			elseif rank >= 200 then
				table.insert(reasons, "High rank in the game's group: " .. role .. " (" .. rank .. ")")
			end
		end
		local okS, inStaff = pcall(plr.IsInGroup, plr, ROBLOX_STAFF_GROUP)
		if okS and inStaff then table.insert(reasons, "Roblox staff member") end
		return reasons, isOwner
	end

	local function showAlert(plr, reasons, isOwner, wasHere, off)
		count += 1
		local card = make("Frame", {BackgroundColor3 = THEME.bg, BorderSizePixel = 0, Size = UDim2.new(1, 0, 0, 120),
			LayoutOrder = count, Parent = holder}, {corner(12)})
		local border = stroke(THEME.accent, 3, 0)
		border.Parent = card
		TweenService:Create(border, TweenInfo.new(0.6, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut, -1, true),
			{Transparency = 0.7}):Play()
		local avatar = make("ImageLabel", {BackgroundColor3 = THEME.panel, BorderSizePixel = 0,
			Position = UDim2.new(0, 14, 0.5, -36), Size = UDim2.fromOffset(72, 72), Parent = card}, {corner(36)})
		task.spawn(function()
			local ok, img = pcall(Players.GetUserThumbnailAsync, Players, plr.UserId,
				Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size150x150)
			if ok and avatar.Parent then avatar.Image = img end
		end)
		make("TextLabel", {BackgroundTransparency = 1, Position = UDim2.fromOffset(100, 8), Size = UDim2.new(1, -150, 0, 26),
			Font = Enum.Font.GothamBlack, Text = isOwner and "GAME OWNER DETECTED" or "ADMIN DETECTED", TextSize = 20,
			TextColor3 = THEME.accent, TextXAlignment = Enum.TextXAlignment.Left, Parent = card})
		make("TextLabel", {BackgroundTransparency = 1, Position = UDim2.fromOffset(100, 34), Size = UDim2.new(1, -110, 0, 18),
			Font = Enum.Font.GothamBold, Text = plr.DisplayName .. "  (@" .. plr.Name .. ")", TextSize = 15,
			TextColor3 = THEME.text, TextXAlignment = Enum.TextXAlignment.Left, TextTruncate = Enum.TextTruncate.AtEnd, Parent = card})
		make("TextLabel", {BackgroundTransparency = 1, Position = UDim2.fromOffset(100, 54), Size = UDim2.new(1, -110, 0, 30),
			Font = Enum.Font.GothamMedium, Text = table.concat(reasons, "\n"), TextSize = 12,
			TextColor3 = Color3.fromRGB(255, 170, 170), TextWrapped = true, TextXAlignment = Enum.TextXAlignment.Left,
			TextYAlignment = Enum.TextYAlignment.Top, Parent = card})
		local status = make("TextLabel", {BackgroundTransparency = 1, Position = UDim2.new(0, 100, 1, -30),
			Size = UDim2.new(1, -110, 0, 26), Font = Enum.Font.Gotham, TextSize = 11, TextColor3 = THEME.subtext,
			TextWrapped = true, TextXAlignment = Enum.TextXAlignment.Left, TextYAlignment = Enum.TextYAlignment.Top, Parent = card,
			Text = (wasHere and "Already in the server" or "Joined") .. " at " .. os.date("%H:%M:%S") .. "  ·  "
				.. (#off > 0 and ("Turned off: " .. table.concat(off, ", ")) or "Nothing was on")})
		local close = make("TextButton", {BackgroundColor3 = THEME.panel, BorderSizePixel = 0, AutoButtonColor = false,
			AnchorPoint = Vector2.new(1, 0), Position = UDim2.new(1, -8, 0, 8), Size = UDim2.fromOffset(28, 28),
			Font = Enum.Font.GothamBold, Text = "×", TextSize = 20, TextColor3 = THEME.text, Parent = card},
			{corner(6), stroke(THEME.accent, 1, 0.3)})
		track(close.Activated:Connect(function() UI.sfx("click"); card:Destroy() end))
		track(Players.PlayerRemoving:Connect(function(p)
			if p == plr and status.Parent then
				status.Text = status.Text .. "  ·  left at " .. os.date("%H:%M:%S")
				status.TextColor3 = THEME.good
			end
		end))
		task.spawn(function()
			for _ = 1, 3 do UI.sfx("error"); task.wait(0.25) end
		end)
	end

	local function check(plr, wasHere)
		if plr == localPlayer or alerted[plr] then return end
		local reasons, isOwner = reasonsFor(plr)
		if #reasons == 0 or not plr.Parent or not gui.Parent then return end
		alerted[plr] = true
		if type(_G.PlayerAnnoyerCleanup) == "function" then return end
		local okOff, off = pcall(_G.BlindButDangerousPanic)
		showAlert(plr, reasons, isOwner, wasHere, okOff and type(off) == "table" and off or {})
	end

	for _, p in ipairs(Players:GetPlayers()) do task.spawn(check, p, true) end
	track(Players.PlayerAdded:Connect(function(p) task.spawn(check, p, false) end))
end

local tpPage = newTab("Teleport", 5, "globe")
do
	local function tpBehind(p)
		local myRoot = rootOf(localPlayer.Character)
		local r = p and p.Parent and rootOf(p.Character)
		if not myRoot then notify("Blind but Dangerous", "You have no character right now", THEME.red); return end
		if not r then notify("Blind but Dangerous", (p and p.DisplayName or "?") .. " has no character right now", THEME.red); return end
		myRoot.AssemblyLinearVelocity = Vector3.zero
		myRoot.CFrame = r.CFrame * CFrame.new(0, 0, 3)
		notify("Blind but Dangerous", "Teleported to " .. p.DisplayName, THEME.good)
	end
	local function alivePlayers()
		local list = {}
		for _, p in ipairs(Players:GetPlayers()) do
			if p ~= localPlayer and isAlive(p.Character) and rootOf(p.Character) then table.insert(list, p) end
		end
		return list
	end

	header(tpPage, "QUICK")
	local qgrid = buttonGrid(tpPage)
	button(tpPage, "Nearest player", function()
		local myRoot = rootOf(localPlayer.Character)
		local best, bestDist
		for _, p in ipairs(alivePlayers()) do
			local d = myRoot and (rootOf(p.Character).Position - myRoot.Position).Magnitude
			if d and (not bestDist or d < bestDist) then best, bestDist = p, d end
		end
		if not best then notify("Blind but Dangerous", "Nobody alive to teleport to"); return end
		tpBehind(best)
	end, qgrid)
	button(tpPage, "Random player", function()
		local list = alivePlayers()
		if #list == 0 then notify("Blind but Dangerous", "Nobody alive to teleport to"); return end
		tpBehind(list[math.random(#list)])
	end, qgrid)
	header(tpPage, "PLAYERS")
	local grid = buttonGrid(tpPage)
	local empty = make("TextLabel", {BackgroundTransparency = 1, Name = "000", LayoutOrder = 0, Font = Enum.Font.Gotham,
		Text = "Nobody else in the server", TextSize = 11, TextColor3 = THEME.subtext,
		TextXAlignment = Enum.TextXAlignment.Left, Parent = grid})
	local headshots, items, btns, alive = {}, {}, {}, true
	onCleanup(function() alive = false end)
	local function label(p) return p.DisplayName .. (isAlive(p.Character) and "" or "  (dead)") end
	local function addButton(i, p)
		local b = button(tpPage, label(p), function() tpBehind(p) end, grid)
		b.LayoutOrder, b.Name = i, string.format("%03d", i)
		local pad = b:FindFirstChildOfClass("UIPadding")
		if pad then pad.PaddingLeft = UDim.new(0, 30) end
		b.TextXAlignment = Enum.TextXAlignment.Left
		local img = make("ImageLabel", {BackgroundColor3 = THEME.bg, BorderSizePixel = 0, AnchorPoint = Vector2.new(0, 0.5),
			Position = UDim2.new(0, -26, 0.5, 0), Size = UDim2.fromOffset(22, 22), Image = headshots[p.UserId] or "",
			Parent = b}, {corner(11), stroke(THEME.stroke, 1, 0)})
		if not headshots[p.UserId] then
			task.spawn(function()
				local ok, url = pcall(Players.GetUserThumbnailAsync, Players, p.UserId,
					Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size48x48)
				if ok then headshots[p.UserId] = url; if img.Parent then img.Image = url end end
			end)
		end
		return b
	end
	task.spawn(function()
		while alive do
			local list = {}
			for _, p in ipairs(Players:GetPlayers()) do if p ~= localPlayer then table.insert(list, p) end end
			table.sort(list, function(a, b) return a.DisplayName:lower() < b.DisplayName:lower() end)
			local same = #list == #items
			if same then for i, p in ipairs(list) do if items[i] ~= p then same = false; break end end end
			if same then
				for i, p in ipairs(items) do btns[i].Text = label(p) end
			else
				for _, b in ipairs(btns) do b:Destroy() end
				table.clear(btns)
				items = list
				for i, p in ipairs(list) do btns[i] = addButton(i, p) end
				empty.Visible = #list == 0
			end
			task.wait(1)
		end
	end)
end

_G.BlindButDangerousCleanup = function()
	for _, c in ipairs(connections) do pcall(function() c:Disconnect() end) end
	table.clear(connections)
	for _, fn in ipairs(cleanupTasks) do pcall(fn) end
	table.clear(cleanupTasks)
	if gui then gui:Destroy() end
	_G.BlindButDangerousCleanup = nil
end
track(closeBtn.Activated:Connect(function() _G.BlindButDangerousCleanup() end))
