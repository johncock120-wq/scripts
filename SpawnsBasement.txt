local connections = {}

local tools = {current = nil, conns = {}, disabled = {}, ranges = {}, errors = {}, windows = {}}
tools.tag = (debug.info(1, "s") or ""):gsub("^[=@]", "")

function tools.log(source, msg, detail)
	table.insert(tools.errors, {time = os.date("%H:%M:%S"), source = source, msg = tostring(msg), detail = detail})
	if tools.onLog then pcall(tools.onLog) end
end

function tools.disable(name, err, when)
	local text = tostring(err)
	local firstLine = text:match("^[^\n]*") or text
	tools.log(name, (when and ("[" .. when .. "] ") or "") .. firstLine, text)
	if tools.disabled[name] then return end
	tools.disabled[name] = true
	for _, c in ipairs(tools.conns[name] or {}) do pcall(function() c:Disconnect() end) end
	local win = tools.windows[name]
	if win then
		win.disabledTool = true
		if tools.hide then pcall(tools.hide, win) end
	end
	if tools.notify then pcall(tools.notify, name .. " broke and was turned off (see error console)") end
end

function tools.run(name, fn)
	task.wait()
	tools.current = name
	tools.conns[name] = tools.conns[name] or {}
	table.insert(tools.ranges, {name = name, first = debug.info(fn, "l") or 0, last = math.huge})
	local ok, err = xpcall(fn, debug.traceback)
	tools.current = nil
	if not ok then tools.disable(name, err, "while loading") end
end

function tools.done()
	local r = tools.ranges[#tools.ranges]
	if r then r.last = debug.info(2, "l") or math.huge end
end

local function track(conn)
	table.insert(connections, conn)
	if tools.current then table.insert(tools.conns[tools.current], conn) end
	return conn
end

track(game:GetService("LogService").MessageOut:Connect(function(msg, mtype)
	if mtype ~= Enum.MessageType.MessageError or tools.tag == "" then return end
	pcall(function()
		if not msg:find(tools.tag, 1, true) then return end
		local line = tonumber(msg:match(tools.tag:gsub("%p", "%%%0") .. ":(%d+):"))
		local owner
		for _, r in ipairs(tools.ranges) do
			if line and line >= r.first and line <= r.last then owner = r.name end
		end
		if owner then tools.disable(owner, msg, "while running") else tools.log("Core", msg) end
	end)
end))

		local p = c.Parent

	local function esc(s) return (s:gsub("&", "&amp;"):gsub("<", "&lt;"):gsub(">", "&gt;")) end

	function tools.highlightLua(src)
		local out, n, i, len = {}, 0, 1, #src
		local prev = ""
		local function push(s, color)
			n += 1
			out[n] = color and ('<font color="' .. color .. '">' .. esc(s) .. '</font>') or esc(s)
		end
		while i <= len do
			local c = src:sub(i, i)
			local e, color
			local longEq = src:match("^%[(=*)%[", i)
			if src:sub(i, i + 1) == "--" then
				local eq = src:match("^%-%-%[(=*)%[", i)
				if eq then
					local _, ce = src:find("]" .. eq .. "]", i + 4 + #eq, true)
					e = ce or len
				else
					e = (src:find("\n", i, true) or len + 1) - 1
				end
				color = C.com
			elseif longEq then
				local _, ce = src:find("]" .. longEq .. "]", i + 2 + #longEq, true)
				e, color = ce or len, C.str
			elseif c == '"' or c == "'" or c == "`" then
				local j = i + 1
				while j <= len do
					local d = src:sub(j, j)
					if d == "\\" then j += 2
					elseif d == c or d == "\n" then break
					else j += 1 end
				end
				e, color = math.min(j, len), C.str
			elseif c:match("%d") or (c == "." and src:sub(i + 1, i + 1):match("%d")) then
				e = select(2, src:find("^0[xXbB][%x_]+", i)) or select(2, src:find("^[%d_]*%.?[%d_]*[eE][%+%-]?%d+", i))
					or select(2, src:find("^[%d_]*%.?[%d_]*", i))
				color = C.num
			elseif c:match("[%a_]") then
				e = select(2, src:find("^[%w_]*", i + 1)) or i
				local word = src:sub(i, e)
				local nextCh = src:match("^%s*(.)", e + 1)
				if prev == "." or prev == ":" then
					color = (nextCh == "(" or nextCh == '"' or nextCh == "{") and C.fn or C.prop
				elseif KEYWORDS[word] then color = C.kw
				elseif LITERALS[word] then color = C.lit
				elseif BUILTINS[word] then color = C.builtin
				elseif nextCh == "(" then color = C.fn
				end
			elseif c:match("%s") then
				e = select(2, src:find("^%s+", i))
			else
				e = i
			end
			local tok = src:sub(i, e)
			push(tok, color)
			local last = tok:match("(%S)%s*$")
			if last then prev = last end
			i = e + 1
		end
		return table.concat(out)
	end

	local MAX_HIGHLIGHT = 60000
	function tools.attachHighlighter(box)
		local label = make("TextLabel", {
			BackgroundTransparency = 1, Position = box.Position, Size = box.Size, AutomaticSize = box.AutomaticSize,
			AnchorPoint = box.AnchorPoint, FontFace = box.FontFace, TextSize = box.TextSize, TextColor3 = box.TextColor3,
			RichText = true, TextWrapped = box.TextWrapped, TextXAlignment = box.TextXAlignment,
			TextYAlignment = box.TextYAlignment, Text = "", ZIndex = box.ZIndex, Parent = box.Parent,
		})
		box.ZIndex += 1
		box.TextTransparency = 1
		local token = 0
		local function refresh()
			token += 1
			local my, text = token, box.Text
			if text == "" then
				label.RichText = false
				label.Text = box.PlaceholderText
				label.TextColor3 = box.PlaceholderColor3
				return
			end
			label.TextColor3 = box.TextColor3
			if #text > MAX_HIGHLIGHT then
				label.RichText = false
				label.Text = text
				return
			end
			task.delay(#text > 8000 and 0.15 or 0, function()
				if my ~= token then return end
				local ok, rich = pcall(tools.highlightLua, text)
				label.RichText = ok
				label.Text = ok and rich or text
			end)
		end
		box:GetPropertyChangedSignal("Text"):Connect(refresh)
		refresh()
		return label
	end
end

local function scaleFromCenter(f, rest, from, to, info, shadow)
	local old = f:FindFirstChild("PopScale")
	if old then old:Destroy() end
	local s = make("UIScale", {Name = "PopScale", Scale = from, Parent = f})
	local size = f.Size
	local w, h = size.X.Offset, size.Y.Offset
	local nv = Instance.new("NumberValue")
	local function apply(v)
		s.Scale = v
		local pos = rest + UDim2.fromOffset(w * (1 - v) / 2, h * (1 - v) / 2)
		f.Position = pos
		if shadow then
			shadow.Size = UDim2.fromOffset((w + 48) * v, (h + 48) * v)
			shadow.Position = pos - UDim2.fromOffset(24 * v, 24 * v)
		end
	end
	nv.Value = from
	apply(from)
	nv.Changed:Connect(apply)
	local tw = TweenService:Create(nv, info, {Value = to})
	tw.Completed:Connect(function() nv:Destroy() end)
	tw:Play()
	return tw, s
end

local function showWindow(win)
	if win.frame.Visible and not win.hiding then return end
	win.hiding = false
	local f = win.frame
	local rest = win.homePos or f.Position
	win.homePos = rest
	f.Visible = true
	local tw, s = scaleFromCenter(f, rest, 0.6, 1, TweenInfo.new(0.7, Enum.EasingStyle.Back, Enum.EasingDirection.Out), win.shadow)
	tw.Completed:Connect(function(state)
		if state == Enum.PlaybackState.Completed and s.Parent then
			s:Destroy()
			f.Position = rest
		end
	end)
	local SOFT = TweenInfo.new(0.5, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)
	if win.shadow then
		win.shadow.ImageTransparency = 1
		tween(win.shadow, {ImageTransparency = 0.4}, TweenInfo.new(0.8, Enum.EasingStyle.Quint, Enum.EasingDirection.Out))
	end

	local border = f:FindFirstChildOfClass("UIStroke")
	if border then
		border.Thickness = 6
		tween(border, {Thickness = 2}, TweenInfo.new(0.9, Enum.EasingStyle.Quint, Enum.EasingDirection.Out))
	end

	if win.logo then
		local old = win.logo:FindFirstChild("IntroScale")
		if old then old:Destroy() end
		local ls = make("UIScale", {Name = "IntroScale", Scale = 0, Parent = win.logo})
		win.logo.Rotation = -140
		task.delay(0.1, function()
			tween(ls, {Scale = 1}, TweenInfo.new(0.6, Enum.EasingStyle.Back, Enum.EasingDirection.Out))
			tween(win.logo, {Rotation = 0}, TweenInfo.new(0.7, Enum.EasingStyle.Back, Enum.EasingDirection.Out))
			task.delay(0.7, function() if ls.Parent then ls:Destroy() end end)
		end)
	end

	if win.titleLabel then
		local label = win.titleLabel
		label.MaxVisibleGraphemes = 0
		local typer = Instance.new("NumberValue")
		typer.Changed:Connect(function(v) label.MaxVisibleGraphemes = math.floor(v) end)
		local tt = TweenService:Create(typer, TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
			{Value = utf8.len(label.Text) or #label.Text})
		tt.Completed:Connect(function()
			label.MaxVisibleGraphemes = -1
			typer:Destroy()
		end)
		task.delay(0.1, function() tt:Play() end)
	end

	win.bodyRest = win.bodyRest or {}
	local items = {}
	for _, c in ipairs(win.body:GetChildren()) do
		if c:IsA("GuiObject") then
			win.bodyRest[c] = win.bodyRest[c] or c.Position
			table.insert(items, c)
		end
	end
	table.sort(items, function(a, b) return win.bodyRest[a].Y.Offset < win.bodyRest[b].Y.Offset end)
	for i, c in ipairs(items) do
		local restPos = win.bodyRest[c]
		c.Position = restPos + UDim2.fromOffset(0, 18)
		task.delay(0.15 + i * 0.04, function()
			if c.Parent then tween(c, {Position = restPos}, SOFT) end
		end)
	end
end

local function hideWindow(win, callback)
	if not win.frame.Visible or win.hiding then
		if callback then callback() end
		return
	end
	win.hiding = true
	local f = win.frame
	local rest = f.Position
	win.homePos = rest
	local _, s = scaleFromCenter(f, rest, 1, 0.6, CLOSE, win.shadow)
	if win.shadow then tween(win.shadow, {ImageTransparency = 1}, CLOSE) end
	task.delay(0.25, function()
		if win.hiding then
			f.Visible = false
			win.hiding = false
		end
		if s.Parent then s:Destroy() end
		f.Position = rest
		if callback then callback() end
	end)
end

local minimizeWindow, restoreWindow
tools.run("Window dock", function()
	local dockBelow = make("Frame", {
		Name = "DockBelow", BackgroundTransparency = 1, Position = UDim2.new(0, 0, 0.5, 52),
		Size = UDim2.new(0, 40, 0, 0), AutomaticSize = Enum.AutomaticSize.Y, ZIndex = 50, Parent = gui,

	local dockAbove = make("Frame", {
		Name = "DockAbove", BackgroundTransparency = 1, AnchorPoint = Vector2.new(0, 1),
		Position = UDim2.new(0, 0, 0.5, -52), Size = UDim2.new(0, 40, 0, 0), AutomaticSize = Enum.AutomaticSize.Y,
		ZIndex = 50, Parent = gui,
	}, {make("UIListLayout", {Padding = UDim.new(0, 16), VerticalAlignment = Enum.VerticalAlignment.Bottom,
		SortOrder = Enum.SortOrder.LayoutOrder})})
	local stackCount = {[dockBelow] = 0, [dockAbove] = 0}
	local tabCount = 0
	local MOVE = TweenInfo.new(0.3, Enum.EasingStyle.Quint, Enum.EasingDirection.InOut)
	local TAB_IN = TweenInfo.new(0.35, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)

	local function offLeft(win)
		local p = win.restorePos
		return UDim2.new(0, -(win.frame.Size.X.Offset + 80), p.Y.Scale, p.Y.Offset)
	end

	restoreWindow = function(win)
		if not win.minimized then return end
		win.minimized = false
		if win.tab then
			local holder, tabBtn = win.tab, win.tabBtn
			win.tab, win.tabBtn = nil, nil
			stackCount[win.tabDock] -= 1
			win.tabDock = nil
			tween(tabBtn, {Position = UDim2.new(0, -40, 0, 0)}, TweenInfo.new(0.2))
			task.delay(0.2, function() holder:Destroy() end)
		end
		local f = win.frame
		f.Position = offLeft(win)
		f.Visible = true
		tween(f, {Position = win.restorePos}, MOVE)
		local s = f:FindFirstChild("PopScale") or make("UIScale", {Name = "PopScale", Scale = 0.6, Parent = f})
		tween(s, {Scale = 1}, OPEN)
		task.delay(0.5, function() if s.Parent and not win.minimized then s:Destroy() end end)
		if win.shadow then
			win.shadow.ImageTransparency = 1
			task.delay(0.35, function()
				if not win.minimized then tween(win.shadow, {ImageTransparency = 0.4}, OPEN) end
			end)
		end
		sfx("slide")
	end

	minimizeWindow = function(win)
		if win.minimized or not win.frame.Visible then return end
		win.minimized = true
		local f = win.frame
		win.restorePos = f.Position
		tween(f, {Position = offLeft(win)}, MOVE)
		local old = f:FindFirstChild("PopScale")
		if old then old:Destroy() end
		tween(make("UIScale", {Name = "PopScale", Parent = f}), {Scale = 0.6}, MOVE)
		if win.shadow then tween(win.shadow, {ImageTransparency = 1}, MOVE) end
		sfx("slide")
		task.delay(0.3, function()
			if win.minimized then f.Visible = false end
		end)

		tabCount += 1
		local h = math.clamp(26 + #win.title * 8, 80, 180)
		local dock = stackCount[dockBelow] <= stackCount[dockAbove] and dockBelow or dockAbove
		stackCount[dock] += 1
		win.tabDock = dock
		local holder = make("Frame", {
			BackgroundTransparency = 1, Size = UDim2.new(0, 40, 0, h),
			LayoutOrder = dock == dockAbove and -tabCount or tabCount, Parent = dock,
		})
		local tabBtn = make("TextButton", {
			BackgroundColor3 = THEME.bg, BorderSizePixel = 0, AutoButtonColor = false,
			Position = UDim2.new(0, -40, 0, 0), Size = UDim2.new(0, 30, 1, 0), Text = "", ZIndex = 50, Parent = holder,
		}, {corner(8), stroke(THEME.accent, 1, 0.3)})
		tools.spikeTab(tabBtn, h)
		make("TextLabel", {
			BackgroundTransparency = 1, AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.5, 4, 0.5, 0),
			Size = UDim2.new(0, h - 14, 0, 18), Rotation = -90,
			Font = Enum.Font.GothamBold, Text = win.title, TextSize = 12, TextColor3 = THEME.accent,
			TextTruncate = Enum.TextTruncate.AtEnd, ZIndex = 51, Parent = tabBtn,
		})
		tween(tabBtn, {Position = UDim2.new(0, -8, 0, 0)}, TAB_IN)
		tabBtn.MouseEnter:Connect(function()
			tween(tabBtn, {BackgroundColor3 = THEME.panelHover, Position = UDim2.new(0, -4, 0, 0)})
		end)
		tabBtn.MouseLeave:Connect(function()
			tween(tabBtn, {BackgroundColor3 = THEME.bg, Position = UDim2.new(0, -8, 0, 0)})
		end)
		tabBtn.Activated:Connect(function() restoreWindow(win) end)
		win.tab, win.tabBtn = holder, tabBtn
	end
	tools.done()
end)

	function t.set(v)
		if v == state then return end
		state = v
		paint(v)
		if onChange then
			local ok, result = pcall(onChange, v)
			if ok and result == false then
				state = not v
				paint(state)
				return
			end
		end
		updateMenu()
		if not loadingSettings and not shuttingDown then sfx(v and "on" or "off") end
		notify(text .. (v and "  ON" or "  OFF"), v and THEME.green or THEME.subtext)
		if not noSave then scheduleSave() end
	end
	function t.get() return state end
	t.key = text
	t.noSave = noSave
	track(row.Activated:Connect(function() t.set(not state) end))
	table.insert(allToggles, t)
	registerItem(text, parent, row, y)
	return t
end

local function sliderRow(parent, y, text, min, max, default, onChange, display)
	local label = sectionLabel(text, y, parent)
	local trackF = make("Frame", {
		BackgroundColor3 = THEME.panel, BorderSizePixel = 0,
		Position = UDim2.new(0, 0, 0, y + 19), Size = UDim2.new(1, 0, 0, 6),
		Active = true, Parent = parent,
	}, {corner(1), stroke(THEME.stroke, 1, 0)})
	local fill = make("Frame", {
		BackgroundColor3 = Color3.new(1, 1, 1), BorderSizePixel = 0, Size = UDim2.new(0, 0, 1, 0), Parent = trackF,
	}, {corner(1), make("UIGradient", {Color = ColorSequence.new(Color3.fromRGB(90, 0, 12), THEME.accent)})})
	local knob = make("Frame", {
		BackgroundColor3 = THEME.text, BorderSizePixel = 0, AnchorPoint = Vector2.new(0.5, 0.5), Rotation = 45,
		Position = UDim2.new(0, 0, 0.5, 0), Size = UDim2.new(0, 11, 0, 11), Parent = trackF,
	}, {corner(1), stroke(THEME.accent, 1, 0)})

	local value
	local s = {}
	function s.set(v)
		value = math.floor(math.clamp(v, min, max) + 0.5)
		local a = (value - min) / (max - min)
		fill.Size = UDim2.new(a, 0, 1, 0)
		knob.Position = UDim2.new(a, 0, 0.5, 0)
		label.Text = text .. "  " .. (display and display(value) or tostring(value))
		if onChange then pcall(onChange, value) end
		scheduleSave()
	end
	function s.get() return value end
	s.key = text
	s.default = default
	s.set(default)
	table.insert(allSliders, s)
	registerItem(text:sub(1, 1) .. text:sub(2):lower(), parent, trackF, y)

	local sliding = false
	local function slideTo(x)
		local a = math.clamp((x - trackF.AbsolutePosition.X) / trackF.AbsoluteSize.X, 0, 1)
		s.set(min + (max - min) * a)
	end
	track(trackF.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			sliding = true
			slideTo(input.Position.X)
		end
	end))
	track(UserInputService.InputChanged:Connect(function(input)
		if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			slideTo(input.Position.X)
		end
	end))
	track(UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			sliding = false
		end
	end))
	return s
end

local function isValidModel(model)
	return model and model.Parent
		and model:FindFirstChild("HumanoidRootPart")
		and model:FindFirstChild("Head")
		and model:FindFirstChildOfClass("Humanoid")

local function findTarget(name)
	if name == "" then return nil end
	local lower = name:lower()
	local prefixHit
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= localPlayer then
			local n, d = plr.Name:lower(), plr.DisplayName:lower()
			if n == lower or d == lower then return plr, plr.Character end
			if not prefixHit and (n:sub(1, #lower) == lower or d:sub(1, #lower) == lower) then
				prefixHit = plr
			end
		end
	end
	if prefixHit then return prefixHit, prefixHit.Character end
	return nil, nil
end

local function reroll()
	spinSpeed = rand(CONFIG.speedRange[1], CONFIG.speedRange[2]) * chaos
	direction = math.random(0, 1) == 0 and -1 or 1
	baseHeight = rand(CONFIG.verticalRange[1], CONFIG.verticalRange[2])
	nextReroll = rand(CONFIG.randomizeEvery[1], CONFIG.randomizeEvery[2]) / chaos
end

local function step(dt)
	if targetPlayer then
		if not targetPlayer.Parent then stop(); task.spawn(flash, "Target left the game"); return end
		targetModel = targetPlayer.Character
	end
	if not isValidModel(targetModel) then return end

	local _, myHum, myHRP = getChar()
	if not myHRP or not myHum or myHum.Health <= 0 then return end

	nextReroll -= dt
	if nextReroll <= 0 then reroll() end

	pulseTimer += dt
	if pulseTimer >= CONFIG.pulseCooldown then
		pulseTimer = 0
		radiusNow = CONFIG.radius + rand(-CONFIG.radiusChaos, CONFIG.radiusChaos) * chaos
		heightNow = baseHeight + rand(-CONFIG.verticalChaos, CONFIG.verticalChaos) * chaos
		if math.random() < 0.1 * chaos then direction = -direction end
	end

	angle += spinSpeed * dt * direction

	local headPos = targetModel.Head.Position
	local pos = headPos + Vector3.new(math.cos(angle) * radiusNow, heightNow, math.sin(angle) * radiusNow)
	local cf = CFrame.lookAt(pos, headPos)
	if CONFIG.spinSelf then
		local tilt = 0.6 * math.min(chaos, 2)
		cf *= CFrame.Angles(rand(-tilt, tilt), rand(-math.pi, math.pi), rand(-tilt, tilt))
	end

	myHum.PlatformStand = true
	myHRP.CFrame = cf
	myHRP.AssemblyLinearVelocity = Vector3.zero
	myHRP.AssemblyAngularVelocity = Vector3.zero

	local cam = Workspace.CurrentCamera
	local tHum = targetModel:FindFirstChildOfClass("Humanoid")
	if cam and tHum and cam.CameraSubject ~= tHum then
		cam.CameraType = Enum.CameraType.Custom
		cam.CameraSubject = tHum
	end
end

	local name = textBox.Text:match("^%s*(.-)%s*$")

	local plr, model = findTarget(name)

	refreshButton()
end

local mimicModel = nil
local mimic = {tracks = {}, noCollide = {}, savedCollide = {}, char = nil, rigWarned = false}
local mimicT, mimicS

function mimic.clearNoCollide()
	for _, nc in ipairs(mimic.noCollide) do pcall(function() nc:Destroy() end) end
	table.clear(mimic.noCollide)
end

function mimic.noCollideWith(targetChar)
	mimic.clearNoCollide()
	local char = localPlayer.Character
	if not char or not targetChar then return end
	for _, mine in ipairs(char:GetChildren()) do
		if mine:IsA("BasePart") then
			for _, theirs in ipairs(targetChar:GetChildren()) do
				if theirs:IsA("BasePart") then
					local nc = Instance.new("NoCollisionConstraint")
					nc.Part0, nc.Part1, nc.Parent = mine, theirs, mine
					table.insert(mimic.noCollide, nc)
				end
			end
		end
	end
end

function mimic.restoreCollide()
	for part, was in pairs(mimic.savedCollide) do
		if part.Parent then part.CanCollide = was end
	end
	table.clear(mimic.savedCollide)
end

function mimic.stopTracks()
	for _, tr in pairs(mimic.tracks) do pcall(function() tr:Stop(0.15) end) end
	table.clear(mimic.tracks)
end

function mimic.setOwnAnimate(enabled)
	local char = localPlayer.Character
	local animate = char and char:FindFirstChild("Animate")
	if animate and (animate:IsA("LocalScript") or animate:IsA("Script")) then
		pcall(function() animate.Disabled = not enabled end)
	end
	if enabled then return end
	local hum = char and char:FindFirstChildOfClass("Humanoid")
	local animator = hum and hum:FindFirstChildOfClass("Animator")
	if not animator then return end
	local ours = {}
	for _, tr in pairs(mimic.tracks) do ours[tr] = true end
	for _, tr in ipairs(animator:GetPlayingAnimationTracks()) do
		if not ours[tr] then pcall(function() tr:Stop(0.1) end) end
	end
end

function mimic.syncAnimations(tHum, hum)
	local tAnimator = tHum:FindFirstChildOfClass("Animator")
	local animator = hum:FindFirstChildOfClass("Animator")
	if not tAnimator or not animator then return end
	local playing = {}
	for _, tt in ipairs(tAnimator:GetPlayingAnimationTracks()) do
		local anim = tt.Animation
		local id = anim and anim.AnimationId
		if id and id ~= "" then
			playing[id] = true
			local mine = mimic.tracks[id]
			if not mine then
				local a = Instance.new("Animation")
				a.AnimationId = id
				local ok, tr = pcall(animator.LoadAnimation, animator, a)
				if ok and tr then
					pcall(function()
						tr.Priority = tt.Priority
						tr.Looped = tt.Looped
					end)
					mimic.tracks[id] = tr
					mine = tr
				end
			end
			if mine then
				if not mine.IsPlaying then mine:Play(0.1, math.max(tt.WeightTarget, 0.01), tt.Speed) end
				if math.abs(mine.Speed - tt.Speed) > 0.01 then mine:AdjustSpeed(tt.Speed) end
				if math.abs(mine.WeightTarget - tt.WeightTarget) > 0.02 then mine:AdjustWeight(tt.WeightTarget, 0.1) end
				if tt.Length > 0 and math.abs(mine.TimePosition - tt.TimePosition) > 0.12 then
					pcall(function() mine.TimePosition = tt.TimePosition end)
				end
			end
		end
	end
	for id, tr in pairs(mimic.tracks) do
		if not playing[id] and tr.IsPlaying then pcall(function() tr:Stop(0.15) end) end
	end
end

function mimic.rootHeight(hum, hrp)
	if hum.RigType == Enum.HumanoidRigType.R6 then return 3 end
	return hum.HipHeight + hrp.Size.Y / 2
end

function mimic.step(syncAnims)
	if not mimicModel or running or flinging then return end
	local plr = Players:GetPlayerFromCharacter(mimicModel)
	if plr then
		if not plr.Parent then
			mimicT.set(false)
			notify("Mimic target left the game", THEME.red)
			return
		end
		if plr.Character and plr.Character ~= mimicModel then
			mimicModel = plr.Character
			mimic.noCollideWith(mimicModel)
		end
	end
	local tHRP = mimicModel:FindFirstChild("HumanoidRootPart")
	local tHum = mimicModel:FindFirstChildOfClass("Humanoid")
	local char, hum, hrp = getChar()
	if not tHRP or not tHum or tHum.Health <= 0 or not hrp or not hum or hum.Health <= 0 then return end

	if mimic.char ~= char then
		mimic.char = char
		table.clear(mimic.tracks)
		table.clear(mimic.savedCollide)
		mimic.setOwnAnimate(false)
		mimic.noCollideWith(mimicModel)
	end

	if hum:GetState() ~= Enum.HumanoidStateType.Physics then hum:ChangeState(Enum.HumanoidStateType.Physics) end
	local dy = mimic.rootHeight(hum, hrp) - mimic.rootHeight(tHum, tHRP)
	hrp.CFrame = tHRP.CFrame * CFrame.new(mimicS.get(), dy, 0)
	hrp.AssemblyLinearVelocity = tHRP.AssemblyLinearVelocity
	hrp.AssemblyAngularVelocity = tHRP.AssemblyAngularVelocity
	for _, part in ipairs(char:GetChildren()) do
		if part:IsA("BasePart") and part.CanCollide then
			if mimic.savedCollide[part] == nil then mimic.savedCollide[part] = true end
			part.CanCollide = false
		end
	end

	if syncAnims then
		if tHum.RigType == hum.RigType then
			mimic.syncAnimations(tHum, hum)
		elseif not mimic.rigWarned then
			mimic.rigWarned = true
			notify("They're " .. tHum.RigType.Name .. ", you're " .. hum.RigType.Name .. ": movement copies, animations can't",
				THEME.accent)
		end
	end
end

tools.run("Server hop", function()
	local row = make("Frame", {
		BackgroundTransparency = 1, Position = UDim2.new(0, 0, 0, 250), Size = UDim2.new(1, 0, 0, 30),
		Parent = generalTab.page,
	})
	local hopping = false

	local function fetch(order)
		local url = string.format(
			"https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=%s&limit=100&excludeFullGames=true",
			game.PlaceId, order)
		local ok, body = pcall(function() return game:HttpGet(url, true) end)
		if not ok or type(body) ~= "string" then return nil, "Couldn't reach the server list" end
		local ok2, data = pcall(HttpService.JSONDecode, HttpService, body)
		if not ok2 or type(data) ~= "table" then return nil, "Bad server list" end
		if type(data.data) ~= "table" then return nil, "Server list rate limited, try again in a bit" end
		return data.data
	end

	local function hop(mode)
		if hopping then notify("Already hopping...", THEME.subtext); return end
		hopping = true
		notify("Finding a server...", THEME.subtext)
		local list, err = fetch(mode == "fullest" and "Desc" or "Asc")
		local options = {}
		for _, s in ipairs(list or {}) do
			local playing, max = tonumber(s.playing), tonumber(s.maxPlayers)
			if type(s.id) == "string" and s.id ~= game.JobId and playing and max and playing < max then
				table.insert(options, s)
			end
		end
		local pick
		if mode == "random" then
			pick = #options > 0 and options[math.random(#options)] or nil
		else
			table.sort(options, function(a, b)
				if mode == "fullest" then return a.playing > b.playing end
				return a.playing < b.playing
			end)
			pick = options[1]
		end
		if not pick then
			hopping = false
			notify(err or "No other server with room", THEME.red)
			return
		end
		notify(string.format("Joining a %d/%d server...", pick.playing, pick.maxPlayers), THEME.green)
		local ok, e = pcall(TeleportService.TeleportToPlaceInstance, TeleportService, game.PlaceId, pick.id, localPlayer)
		if not ok then notify("Teleport failed: " .. tostring(e), THEME.red) end
		task.delay(10, function() hopping = false end)
	end

	for i, def in ipairs({{"Hop: smallest", "smallest"}, {"Hop: fullest", "fullest"}, {"Hop: random", "random"}}) do
		local b = make("TextButton", {
			Position = UDim2.new((i - 1) / 3, (i - 1) * 2, 0, 0), Size = UDim2.new(1 / 3, -4, 1, 0),
			Text = def[1], Parent = row,
		}, {corner(8)})
		styleButton(b)
		b.TextSize = 12
		track(b.Activated:Connect(function() task.spawn(hop, def[2]) end))
	end
	registerItem("Server hop", generalTab.page, row, 250)
	tools.done()
end)

tools.run("Anti-fling / anti-void", function()
	local antiFlingT = toggleRow(charTab.page, 282, "Anti-fling")
	local antiVoidT = toggleRow(charTab.page, 318, "Anti-void")
	local safeCF, safeAt, lastVel, lastRescue, voidTime = nil, 0, nil, 0, 0
	local rayParams = RaycastParams.new()
	rayParams.FilterType = Enum.RaycastFilterType.Exclude

	local function movingMyself()
		return flyT.get() or running or flinging or mimicT.get() or stalkT.get()
	end

	local function rescue(why)
		local _, hum, hrp = getChar()
		if not hrp or os.clock() - lastRescue < 0.5 then return end
		lastRescue = os.clock()
		hrp.AssemblyLinearVelocity = Vector3.zero
		hrp.AssemblyAngularVelocity = Vector3.zero
		if safeCF then hrp.CFrame = safeCF end
		if hum then pcall(hum.ChangeState, hum, Enum.HumanoidStateType.GettingUp) end
		lastVel, voidTime = Vector3.zero, 0
		notify(why, THEME.accent)
	end

	track(localPlayer.CharacterAdded:Connect(function()
		safeCF, lastVel, voidTime = nil, nil, 0
	end))

	track(RunService.Heartbeat:Connect(function(dt)
		if not antiFlingT.get() and not antiVoidT.get() then lastVel = nil; return end
		local char, hum, hrp = getChar()
		if not hrp or not hum or hum.Health <= 0 then lastVel = nil; return end
		local vel, spin = hrp.AssemblyLinearVelocity, hrp.AssemblyAngularVelocity
		local mine = movingMyself()

		if not mine and hum.FloorMaterial ~= Enum.Material.Air and vel.Magnitude < 60 and spin.Magnitude < 10
			and os.clock() - safeAt > 0.2 then
			safeCF, safeAt = hrp.CFrame, os.clock()
		end

		if antiFlingT.get() and not mine and lastVel then
			local change = vel - lastVel
			local flatChange = Vector3.new(change.X, 0, change.Z).Magnitude
			local jumpVel = hum.UseJumpPower and hum.JumpPower or math.sqrt(2 * Workspace.Gravity * hum.JumpHeight)
			local flatSpeed = Vector3.new(vel.X, 0, vel.Z).Magnitude
			if spin.Magnitude > 60 or flatChange > 200 or change.Y > jumpVel + 150 or flatSpeed > 600 then
				rescue("Anti-fling: stopped a fling")
				return
			end
		end
		lastVel = vel

		if antiVoidT.get() and not mine then
			local kill = Workspace.FallenPartsDestroyHeight
			if hrp.Position.Y < kill + 60 then
				rescue("Anti-void: caught you before the void")
				return
			end
			if vel.Y < -60 then
				rayParams.FilterDescendantsInstances = {char}
				local depth = math.min(hrp.Position.Y - kill, 15000)
				local down = Workspace:Raycast(hrp.Position, Vector3.new(0, -depth, 0), rayParams)
				local along = not down and Workspace:Raycast(hrp.Position, vel.Unit * math.min(depth * 1.5, 15000), rayParams)
				if down or along then
					voidTime = 0
				else
					voidTime += dt
					if voidTime > 0.35 then rescue("Anti-void: caught you falling into the void") end
				end
			else
				voidTime = 0
			end
		end
	end))
	tools.done()
end)

	local function buildScript(entry)
		local call = entry.last
		local lines = {"local args = {"}
		for i = 1, call.args.n do
			table.insert(lines, "\t" .. ser(call.args[i], 1, "\t") .. ",")
		end
		table.insert(lines, "}")
		local path = instPath(entry.remote)
		local n = call.args.n
		if entry.dir == "out" then
			table.insert(lines, path .. ":" .. call.method .. "(unpack(args, 1, " .. n .. "))")
		elseif entry.dir == "in" then
			table.insert(lines, "firesignal(" .. path .. ".OnClientEvent, unpack(args, 1, " .. n .. "))")
		elseif entry.remote:IsA("BindableEvent") then
			table.insert(lines, path .. ":Fire(unpack(args, 1, " .. n .. "))")
		else
			table.insert(lines, "firesignal(" .. path .. ".Event, unpack(args, 1, " .. n .. "))")
		end
		return table.concat(lines, "\n")
	end

	local ROBLOX_CONTAINERS = {"RobloxReplicatedStorage", "CoreGui", "CorePackages"}

	local function isRobloxRemote(remote)

	spy.history = {}

	local function describeBuffer(remote, buf)
		local len = buffer.len(buf)
		local first = len > 0 and buffer.readu8(buf, 0) or nil
		local pname
		if first then
			local okA, attrs = pcall(remote.GetAttributes, remote)
			if okA then
				for k, v in pairs(attrs) do
					if v == first then pname = k; break end
				end
			end
		end
		local hex, txt = {}, {}
		for i = 0, math.min(len, 512) - 1 do
			local b = buffer.readu8(buf, i)
			hex[#hex + 1] = string.format("%02X", b)
			txt[#txt + 1] = (b >= 32 and b < 127) and string.char(b) or "."
		end
		return string.format("buffer(%d bytes)%s\n      hex:  %s%s\n      text: %s", len,
			first and ("  first packet id " .. first .. (pname and (" = " .. pname) or "")) or "",
			table.concat(hex, " "), len > 512 and " ..." or "", table.concat(txt))
	end

	local function argsText(remote, args)
		local lines = {}
		for i = 1, args.n do
			local v = args[i]
			local okS, s = pcall(function()
				if typeof(v) == "buffer" then
					local raw = describeBuffer(remote, v)
					local okD, decoded = pcall(packetDecoder.decode, remote, v, i == 1 and args[2] or nil)
					if okD and decoded then
						return raw .. "\n      decoded:\n        " .. table.concat(decoded, "\n        ")
					end
					return raw
				end
				return ser(v, 1, "    ")
			end)
			lines[#lines + 1] = "    [" .. i .. "] " .. (okS and s or ("<" .. typeof(v) .. ">"))
		end
		return #lines > 0 and table.concat(lines, "\n") or "    (no arguments)"
	end

	local function onRemoteCall(remote, method, args, dir)

		if dir ~= "out" and not spy.incoming then return end
		local okP, path = pcall(instPath, remote)
		table.insert(spy.history, string.format("[%s] %s  %s  %s:%s\n%s",
			os.date("%H:%M:%S"), dir == "out" and "OUT ->" or dir == "bind" and "BIND <>" or "IN  <-", remote.ClassName,
			okP and path or remote.Name, method, argsText(remote, args)))
		if #spy.history > HISTORY_MAX then table.remove(spy.history, 1) end
		spySerial += 1
		local map = spyEntries[dir]
		local e = map[remote]
		if not e then

				Parent = row,

			makeIcon(dir == "out" and "arrow-up-right" or "arrow-down-left", 12, chip, Color3.new(1, 1, 1), {
				AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.5, 0, 0.5, 0),
			})
			make("ImageLabel", {
				BackgroundTransparency = 1, Image = CLASS_IMAGES[remote.ClassName] or CLASS_IMAGES.RemoteEvent,
				AnchorPoint = Vector2.new(0, 0.5), Position = UDim2.new(0, 30, 0.5, 0), Size = UDim2.new(0, 16, 0, 16),
				Parent = row,
			})
			make("TextLabel", {

	local function installHook()
		if spyHooked or not canHook then return end
		spyHooked = true
		local wrap = typeof(newcclosure) == "function" and newcclosure or function(f) return f end
		local state = spy
		local oldNamecall
		oldNamecall = hookmetamethod(game, "__namecall", wrap(function(self, ...)
			if state.enabled and not state.dead and typeof(self) == "Instance" then
				local method = getnamecallmethod()
				if (method == "FireServer" or method == "InvokeServer") and REMOTE_CLASSES[self.ClassName]
					and not (hasCheckcaller and checkcaller()) then
					if state.blocked[self] then return nil end
					local args = table.pack(...)
					task.defer(onRemoteCall, self, method, args, "out")
				end
			end
			return oldNamecall(self, ...)
		end))
	end

	local incomingHooked = {}
	local callbackWrapped = setmetatable({}, {__mode = "k"})
	local function wrapCallback(inst, prop, dir)
		local okG, current = pcall(getcallbackvalue, inst, prop)
		if not okG or type(current) ~= "function" or current == callbackWrapped[inst] then return end
		local original = current
		local wrapper
		wrapper = function(...)
			local args = table.pack(...)
			task.defer(onRemoteCall, inst, prop, args, dir)
			return original(...)
		end
		if pcall(function() inst[prop] = wrapper end) then callbackWrapped[inst] = wrapper end
	end

	local function watchIncoming(inst)
		if incomingHooked[inst] or isRobloxRemote(inst) then return end
		local cls = inst.ClassName
		if cls == "RemoteEvent" or cls == "UnreliableRemoteEvent" then
			incomingHooked[inst] = true
			track(inst.OnClientEvent:Connect(function(...)
				onRemoteCall(inst, "OnClientEvent", table.pack(...), "in")
			end))
		elseif cls == "BindableEvent" then
			incomingHooked[inst] = true
			track(inst.Event:Connect(function(...)
				onRemoteCall(inst, "Event", table.pack(...), "bind")
			end))
		elseif (cls == "RemoteFunction" or cls == "BindableFunction") and canWrapCallbacks then
			incomingHooked[inst] = true
			wrapCallback(inst, cls == "RemoteFunction" and "OnClientInvoke" or "OnInvoke",
				cls == "RemoteFunction" and "in" or "bind")
		end
	end
	local incomingReady = false
	local function installIncoming()
		if incomingReady then return end
		incomingReady = true
		for i, d in ipairs(game:GetDescendants()) do
			if WATCH_CLASSES[d.ClassName] then pcall(watchIncoming, d) end
			if i % 4000 == 0 then task.wait() end
		end
		track(game.DescendantAdded:Connect(function(d)
			if WATCH_CLASSES[d.ClassName] then task.defer(pcall, watchIncoming, d) end
		end))
		if canWrapCallbacks then
			task.spawn(function()
				while not spy.dead and not shuttingDown do
					task.wait(2)
					if spy.enabled then
						for inst in pairs(incomingHooked) do
							local cls = inst.ClassName
							if inst.Parent and (cls == "RemoteFunction" or cls == "BindableFunction") then
								pcall(wrapCallback, inst, cls == "RemoteFunction" and "OnClientInvoke" or "OnInvoke",
									cls == "RemoteFunction" and "in" or "bind")
							end
						end
					end
				end
			end)
		end
	end

	function replay.dispatch(e, argv, n)
		local r = e.remote
		if e.dir == "out" then
			if e.last and e.last.method == "InvokeServer" then
				task.spawn(function() pcall(function() r:InvokeServer(unpack(argv, 1, n)) end) end)
			else
				r:FireServer(unpack(argv, 1, n))
			end
		elseif e.dir == "in" then
			if typeof(firesignal) == "function" then firesignal(r.OnClientEvent, unpack(argv, 1, n)) end
		elseif r:IsA("BindableEvent") then
			r:Fire(unpack(argv, 1, n))
		elseif typeof(firesignal) == "function" then
			firesignal(r.Event, unpack(argv, 1, n))
		end
	end

	function replay.runScript()
		if typeof(loadstring) ~= "function" then notify("Your executor has no loadstring", THEME.red); return false end
		local fn, err = loadstring(fireBox.Text, "=RemoteReplay")
		if not fn then notify("Script error: " .. tostring(err):gsub("^.-:%d+: ", ""), THEME.red); return false end
		local ok, runErr = pcall(fn)
		if not ok then notify("Fire error: " .. tostring(runErr), THEME.red); return false end
		return true
	end

	function replay.fuzz(e)
		local base = e.last and e.last.args or table.pack()
		local deep = {}
		local cur = deep
		for _ = 1, 40 do cur.n = {}; cur = cur.n end
		local sets = {
			{n = 0},
			table.pack(0 / 0), table.pack(math.huge), table.pack(-math.huge),
			table.pack(1e18), table.pack(-1e18), table.pack(2 ^ 53),
			table.pack(""), table.pack(string.rep("A", 20000)),
			table.pack(true), table.pack(false), table.pack(deep),
			table.pack(Vector3.new(1e9, 1e9, 1e9)),
		}
		for i = 1, base.n or 0 do
			local t = typeof(base[i])
			if t == "number" or t == "string" then
				for _, bad in ipairs({0 / 0, math.huge, -1e18, string.rep("Z", 10000)}) do
					local copy = table.pack(table.unpack(base, 1, base.n))
					copy[i] = bad
					table.insert(sets, copy)
				end
			end
		end
		local fired = 0
		for _, set in ipairs(sets) do
			if not e.remote.Parent then break end
			if pcall(replay.dispatch, e, set, set.n or #set) then fired += 1 end
			task.wait(0.05)
		end
		notify("Fuzzed " .. e.remote.Name .. " with " .. fired .. " payloads (watch for a reaction)", THEME.accent)
	end

	end

local function instPathOf(inst)
	if inst == game then return "game" end
	local segs = {}
	local cur = inst
	while cur and cur.Parent ~= game do
		local name = cur.Name
		if name:match("^[%a_][%w_]*$") then
			table.insert(segs, 1, "." .. name)
		else
			table.insert(segs, 1, '["' .. name:gsub("\\", "\\\\"):gsub('"', '\\"') .. '"]')
		end
		cur = cur.Parent
	end
	if not cur then return "nil --[[ parented to nil: " .. inst.Name .. " ]]" end
	if cur == Workspace then return "workspace" .. table.concat(segs) end
	return 'game:GetService("' .. cur.ClassName .. '")' .. table.concat(segs)
end

	local function numStr(x)
		return (string.format("%.3f", x):gsub("%.?0+$", ""))

	local function editText(v)
		local t = typeof(v)
		if t == "string" then return v
		elseif t == "number" then return numStr(v)
		elseif t == "Vector3" then return numStr(v.X) .. ", " .. numStr(v.Y) .. ", " .. numStr(v.Z)
		elseif t == "Vector2" then return numStr(v.X) .. ", " .. numStr(v.Y)
		elseif t == "CFrame" then local p = v.Position; return numStr(p.X) .. ", " .. numStr(p.Y) .. ", " .. numStr(p.Z)
		elseif t == "Color3" then
			return math.floor(v.R * 255 + 0.5) .. ", " .. math.floor(v.G * 255 + 0.5) .. ", " .. math.floor(v.B * 255 + 0.5)
		elseif t == "UDim2" then
			return numStr(v.X.Scale) .. ", " .. v.X.Offset .. ", " .. numStr(v.Y.Scale) .. ", " .. v.Y.Offset
		elseif t == "EnumItem" or t == "BrickColor" then return v.Name
		end
		return fmt(v)
	end

	local function parseValue(old, text)
		local t = typeof(old)
		if t == "string" then return text end
		local n = {}
		for s in text:gmatch("[^,%s]+") do table.insert(n, tonumber(s)) end
		if t == "number" then return tonumber(text)
		elseif t == "Vector3" and #n >= 3 then return Vector3.new(n[1], n[2], n[3])
		elseif t == "Vector2" and #n >= 2 then return Vector2.new(n[1], n[2])
		elseif t == "CFrame" and #n >= 3 then return CFrame.new(n[1], n[2], n[3]) * old.Rotation
		elseif t == "Color3" and #n >= 3 then
			return Color3.fromRGB(math.clamp(n[1], 0, 255), math.clamp(n[2], 0, 255), math.clamp(n[3], 0, 255))
		elseif t == "UDim2" and #n >= 4 then return UDim2.new(n[1], n[2], n[3], n[4])
		elseif t == "BrickColor" then
			local ok, bc = pcall(BrickColor.new, text)
			return ok and bc or nil
		elseif t == "EnumItem" then
			for _, item in ipairs(old.EnumType:GetEnumItems()) do
				if item.Name:lower() == text:lower():match("^%s*(.-)%s*$") then return item end
			end
		end
		return nil
	end

	local function renderProps()
		if not selected then return end
		local inst = selected
		if not isAlive(inst) then
			clearProps()
			placeholder.Visible = true
			placeholder.Text = "(this object was destroyed)"
			propState.inst = nil
			selBox.Adornee = nil
			return
		end
		local keys = {}
		for _, p in ipairs(PROPS) do
			local ok, v = pcall(function() return inst[p] end)
			if ok and v ~= nil then table.insert(keys, p) end
		end
		local attrNames = {}
		local okA, attrs = pcall(inst.GetAttributes, inst)
		if okA then
			for name in pairs(attrs) do table.insert(attrNames, name) end
			table.sort(attrNames)
		end
		local sig = table.concat(keys, "|") .. "#" .. table.concat(attrNames, "|")

		if propState.inst == inst and propState.sig == sig then
			for _, entry in ipairs(propState.rows) do entry.update() end
			return
		end
		propState.inst, propState.sig = inst, sig
		clearProps()
		placeholder.Visible = false
		local order = 1

		if inst:IsA("LuaSourceContainer") then
			local viewBtn = make("TextButton", {
				Size = UDim2.new(1, -6, 0, 28), LayoutOrder = order, Text = "View source", Parent = props,
			}, {corner(6)})
			styleButton(viewBtn, THEME.accent)
			viewBtn.Text = canEdit(inst) and "Open script  (editable)" or "Open script  (read-only)"
			viewBtn.Activated:Connect(function() openScript(inst) end)
			order += 1
		end

		header("PROPERTIES  (click a value to edit, Enter to apply)", order); order += 1
		for _, p in ipairs(keys) do
			propRow(order, p, function() return inst[p] end,
				not READ_ONLY[p] and function(val) inst[p] = val end or nil)
			order += 1
		end

		if #attrNames > 0 then
			header("ATTRIBUTES", order); order += 1
			for _, name in ipairs(attrNames) do
				propRow(order, name, function() return inst:GetAttribute(name) end,
					function(val) inst:SetAttribute(name, val) end)
				order += 1
			end
		end

		header("INFO", order); order += 1
		propRow(order, "Children", function()
			local ok, kids = pcall(inst.GetChildren, inst)
			return ok and #kids or "?"
		end); order += 1
		make("TextLabel", {
			BackgroundTransparency = 1, Size = UDim2.new(1, -6, 0, 0), AutomaticSize = Enum.AutomaticSize.Y,
			LayoutOrder = order, Font = Enum.Font.Code, Text = instPathOf(inst), TextSize = 12,
			TextColor3 = THEME.text, TextWrapped = true, TextXAlignment = Enum.TextXAlignment.Left, Parent = props,
		})
		header("Edits only change things on your screen.", order + 1)
	end

	local function sortedChildren(inst)
		local ok, kids = pcall(inst.GetChildren, inst)

		for _, k in ipairs(kids) do
			local okN, n = pcall(function() return k.Name end)
			if okN then table.insert(named, {k, n:lower()}) end
		end
		table.sort(named, function(a, b) return a[2] < b[2] end)
		local out = {}
		for i, pair in ipairs(named) do out[i] = pair[1] end
		return out
	end

tools.run("Script dump", function()
	local dumping = false
	local SCRIPT_CLASSES = {LocalScript = true, ModuleScript = true, Script = true}
	local REMOTE_CLASSES = {RemoteEvent = true, UnreliableRemoteEvent = true, RemoteFunction = true,
		BindableEvent = true, BindableFunction = true}

	local function roots()
		local list = {}
		for _, name in ipairs({"Workspace", "ReplicatedFirst", "ReplicatedStorage", "StarterPlayer", "StarterGui",
			"StarterPack", "Lighting", "SoundService", "Chat", "Teams", "TextChatService", "Players"}) do
			local ok, svc = pcall(game.FindService, game, name)
			if ok and svc then table.insert(list, svc) end
		end
		return list
	end

	local function safeName(s)
		s = tostring(s):gsub("[^%w%._%- ]", "_")
		return #s > 60 and s:sub(1, 60) or s
	end

	local function dump()
		if dumping then notify("Already dumping...", THEME.subtext); return end
		if typeof(writefile) ~= "function" or typeof(makefolder) ~= "function" then
			notify("Your executor can't write files/folders", THEME.red); return
		end
		local age = Workspace.DistributedGameTime
		if age < 30 then
			notify(string.format("Game still loading - wait %d more seconds", math.ceil(30 - age)), THEME.red)
			return
		end
		dumping = true
		local canDecompile = typeof(decompile) == "function"
		local hashOf = typeof(getscripthash) == "function" and getscripthash or nil
		local folder = "Dump_" .. game.PlaceId .. "_" .. os.date("%Y-%m-%d_%H-%M-%S")
		pcall(makefolder, folder)
		pcall(makefolder, folder .. "/scripts")
		notify("Collecting scripts and remotes...", THEME.accent)

		local scripts, remotes, visited = {}, {}, 0
		local function walk(inst)
			local ok, kids = pcall(inst.GetChildren, inst)
			if not ok then return end
			for _, c in ipairs(kids) do
				local okC, cls = pcall(function() return c.ClassName end)
				if okC then
					if SCRIPT_CLASSES[cls] then table.insert(scripts, c)
					elseif REMOTE_CLASSES[cls] then table.insert(remotes, c) end
				end
				visited += 1
				if visited % 2000 == 0 then task.wait() end
				walk(c)
			end
		end
		for _, r in ipairs(roots()) do walk(r) end
		if typeof(getnilinstances) == "function" then
			local ok, nils = pcall(getnilinstances)
			if ok and type(nils) == "table" then
				for _, c in ipairs(nils) do
					local okC, cls = pcall(function() return c.ClassName end)
					if okC and SCRIPT_CLASSES[cls] then table.insert(scripts, c)
					elseif okC and REMOTE_CLASSES[cls] then table.insert(remotes, c) end
				end
			end
		end

		table.sort(remotes, function(a, b) return instPathOf(a) < instPathOf(b) end)
		local rlines = {"Remotes & bindables - " .. os.date("%Y-%m-%d %H:%M:%S") .. "  |  PlaceId " .. game.PlaceId,
			"Total: " .. #remotes, ""}
		for _, r in ipairs(remotes) do
			local okA, attrs = pcall(r.GetAttributes, r)
			local attrText = ""
			if okA and next(attrs) then
				local parts = {}
				for k, v in pairs(attrs) do table.insert(parts, k .. "=" .. tostring(v)) end
				table.sort(parts)
				attrText = "   attrs: " .. table.concat(parts, ", ")
			end
			table.insert(rlines, string.format("%-22s %s%s", r.ClassName, instPathOf(r), attrText))
		end
		pcall(writefile, folder .. "/_remotes.txt", table.concat(rlines, "\n"))

		local index = {"Scripts - " .. os.date("%Y-%m-%d %H:%M:%S") .. "  |  PlaceId " .. game.PlaceId, ""}
		local used, byHash = {}, {}
		local done, failed, server = 0, 0, 0
		for i, scr in ipairs(scripts) do
			local path = instPathOf(scr)
			local isServer = scr.ClassName == "Script"
			if isServer then
				local okRC, rc = pcall(function() return scr.RunContext end)
				isServer = not (okRC and rc == Enum.RunContext.Client)
			end
			local status
			if isServer then
				server += 1
				status = "server script (source never reaches the client)"
			else
				local src, why
				local okH, hash = false, nil
				if hashOf then okH, hash = pcall(hashOf, scr) end
				if okH and hash and byHash[hash] then
					src = byHash[hash]
				elseif canDecompile then
					local okD, out = pcall(decompile, scr)
					if okD and type(out) == "string" and #out > 0 then src = out else why = tostring(out) end
					if src and okH and hash then byHash[hash] = src end
				else
					why = "no decompile()"
				end
				local base = safeName(path:gsub('^game:GetService%("([%w]+)"%)', "%1"):gsub('%["', "."):gsub('"%]', ""))
				local name = base
				local n = 1
				while used[name] do n += 1; name = base .. "_" .. n end
				used[name] = true
				if src then
					pcall(writefile, folder .. "/scripts/" .. name .. ".lua",
						"-- " .. scr.ClassName .. "  " .. path .. "\n\n" .. src)
					done += 1
					status = "scripts/" .. name .. ".lua"
				else
					failed += 1
					status = "FAILED: " .. tostring(why):match("^[^\n]*")
				end
				task.wait(0.03)
			end
			table.insert(index, string.format("%-13s %s\n              -> %s", scr.ClassName, path, status))
			if i % 25 == 0 then notify(string.format("Decompiling %d / %d scripts...", i, #scripts), THEME.subtext) end
		end
		table.insert(index, 2, string.format("Decompiled: %d   Failed: %d   Server (skipped): %d", done, failed, server))
		pcall(writefile, folder .. "/_index.txt", table.concat(index, "\n"))
		dumping = false
		notify(string.format("Saved %d scripts + %d remotes to %s", done, #remotes, folder), THEME.green)
	end

	buttonRow(utilityTab.page, 144, "Save scripts + remotes", function() dump() end)
	tools.done()
end)

				return
			end
			if not here and os.clock() - (wrongGameArmed[g] or 0) > 3 then

				notify("You're not in " .. g.name .. " - click again to load it anyway", THEME.accent)
				return
			end

			notify("Loading " .. g.name .. (hasLocal and " (local file)" or " (GitHub)") .. "...", THEME.accent)
			local okGet, src = pcall(function()
				if hasLocal then return readfile(g.file) end
				return game:HttpGet(g.url, true)
			end)
			if not okGet or type(src) ~= "string" or src == "" then
				loadingGame = false
				notify("Couldn't " .. (hasLocal and "read " or "download ") .. g.name .. ": " .. tostring(src), THEME.red)
				return
			end

				else
					notify(g.name .. " crashed: " .. tostring(runErr), THEME.red)
				end
			end)
		end)
	end
	tools.done()
end)

			for _ = 1, 3 do
				sfx("error")
				task.wait(0.25)
			end
		end)
	end

	local function check(plr, wasHere)
		if plr == localPlayer or alerted[plr] then return end
		local reasons = {}
		local isOwner = false
		if game.CreatorType == Enum.CreatorType.User and plr.UserId == game.CreatorId then
			isOwner = true
			table.insert(reasons, "Owns this game")
		elseif game.CreatorType == Enum.CreatorType.Group then
			local okR, rank = pcall(plr.GetRankInGroup, plr, game.CreatorId)
			local okN, role = pcall(plr.GetRoleInGroup, plr, game.CreatorId)
			rank = okR and rank or 0
			role = okN and role or "?"
			local lowerRole = role:lower()
			local staffWord = false
			for _, w in ipairs(STAFF_WORDS) do
				if lowerRole:find(w, 1, true) then staffWord = true; break end
			end
			if rank == 255 then
				isOwner = true
				table.insert(reasons, "Owns this game (owner of the game's group: " .. role .. ")")
			elseif rank >= 254 then
				table.insert(reasons, "Top rank in the game's group: " .. role .. " (" .. rank .. ")")
			elseif rank > 0 and staffWord then
				table.insert(reasons, "Staff role in the game's group: " .. role .. " (" .. rank .. ")")
			elseif rank >= 200 then
				table.insert(reasons, "High rank in the game's group: " .. role .. " (" .. rank .. ")")
			end
		end
		local okS, inStaff = pcall(plr.IsInGroup, plr, ROBLOX_STAFF_GROUP)
		if okS and inStaff then
			table.insert(reasons, "Roblox staff member")
		end
		if #reasons > 0 and plr.Parent and not shuttingDown then
			alerted[plr] = true
			tools.adminsHere[plr] = true
			local turnedOff = disableEverything()
			showAlert(plr, reasons, wasHere, isOwner, turnedOff)
		end
	end

	for _, plr in ipairs(Players:GetPlayers()) do
		task.spawn(check, plr, true)
	end
	track(Players.PlayerAdded:Connect(function(plr) task.spawn(check, plr, false) end))
	track(Players.PlayerRemoving:Connect(function(plr)
		tools.adminsHere[plr] = nil
		local status = cards[plr]
		if status and status.Parent then
			status.Text = status.Text .. "  ·  left at " .. os.date("%H:%M:%S")
			status.TextColor3 = Color3.fromRGB(80, 220, 120)
		end
	end))
	tools.done()
end)

local menuOpen = false
local slideMenu, bootOpen
tools.run("Menu slide", function()
	local menuSettled = false
	local slideToken = 0
	local homePos = mainWin.frame.Position
	local SLIDE_IN = TweenInfo.new(0.35, Enum.EasingStyle.Quint, Enum.EasingDirection.Out)
	local SLIDE_OUT = TweenInfo.new(0.3, Enum.EasingStyle.Quint, Enum.EasingDirection.In)

	local handle = make("TextButton", {
		BackgroundColor3 = THEME.bg, BorderSizePixel = 0, AutoButtonColor = false,
		AnchorPoint = Vector2.new(0, 0.5), Position = UDim2.new(0, -40, 0.5, 0), Size = UDim2.new(0, 30, 0, 70),
		Font = Enum.Font.GothamBlack, Text = "", TextSize = 22, TextColor3 = THEME.accent, Parent = gui,
	}, {corner(8), stroke(THEME.accent, 1, 0.3), make("UIPadding", {PaddingLeft = UDim.new(0, 8)})})
	hoverable(handle, THEME.bg, THEME.panelHover)
	tools.spikeTab(handle, 70)
	makeIcon("chevron-right", 18, handle, THEME.accent, {
		AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.5, 0, 0.5, 0),
	})

	local function offscreenPos()
		return UDim2.new(0, -(WIN_W + 80), homePos.Y.Scale, homePos.Y.Offset)
	end

	slideMenu = function(open, callback)
		local f = mainWin.frame
		if open ~= menuOpen then sfx("slide") end
		slideToken += 1
		local my = slideToken
		if open then
			if not f.Visible then
				f.Position = offscreenPos()
				f.Visible = true
			end
			menuSettled = false
			tween(f, {Position = homePos}, SLIDE_IN)
			tween(handle, {Position = UDim2.new(0, -40, 0.5, 0)}, SLIDE_OUT)
			task.delay(0.35, function()
				if slideToken == my then menuSettled = true end
			end)
		else
			if menuOpen and menuSettled then homePos = f.Position end
			menuSettled = false
			tween(f, {Position = offscreenPos()}, SLIDE_OUT)
			if not callback then
				tween(handle, {Position = UDim2.new(0, -8, 0.5, 0)}, SLIDE_IN)
			end
			task.delay(0.3, function()
				if slideToken ~= my then return end
				f.Visible = false
				if callback then callback() end
			end)
		end
		menuOpen = open
	end

	bootOpen = function()
		local f = mainWin.frame
		slideToken += 1
		local my = slideToken
		homePos = UDim2.new(0.5, -WIN_W / 2, 0.5, -WIN_H / 2)
		f.Visible = true
		local tw, s = scaleFromCenter(f, homePos, 0.6, 1, TweenInfo.new(0.7, Enum.EasingStyle.Back, Enum.EasingDirection.Out), mainWin.shadow)
		tw.Completed:Connect(function()
			if s.Parent then s:Destroy() end
			f.Position = homePos
			if slideToken == my then menuSettled = true end
		end)
		mainWin.shadow.ImageTransparency = 1
		tween(mainWin.shadow, {ImageTransparency = 0.4}, TweenInfo.new(0.8, Enum.EasingStyle.Quint, Enum.EasingDirection.Out))
		menuOpen, menuSettled = true, false
	end

	track(handle.Activated:Connect(function() slideMenu(true) end))
	track(mainWin.minBtn.Activated:Connect(function() slideMenu(false) end))
	tools.done()
end)

local SETTINGS_FILE = "PlayerAnnoyerSettings.json"
local canSave = typeof(writefile) == "function" and typeof(readfile) == "function" and typeof(isfile) == "function"
local settingsReady = false
local saveToken = 0
local DEFAULT_CHAOS_ALPHA = (1 - CONFIG.chaosMin) / (CONFIG.chaosMax - CONFIG.chaosMin)

local function saveNow()
	if not canSave or not settingsReady then return end
	local data = {toggles = {}, sliders = {}, chaos = chaos, tab = currentTab and currentTab.name}
	for _, t in ipairs(allToggles) do
		if not t.noSave then data.toggles[t.key] = t.get() end
	end
	for _, s in ipairs(allSliders) do
		data.sliders[s.key] = s.get()
	end
	pcall(writefile, SETTINGS_FILE, HttpService:JSONEncode(data))
end

scheduleSave = function()
	if not settingsReady or shuttingDown or not canSave then return end
	saveToken += 1
	local my = saveToken
	task.delay(1, function()
		if my == saveToken and not shuttingDown then saveNow() end
	end)
end

local function loadSettings()
	if not canSave then return nil end
	local ok, data = pcall(function()
		if not isfile(SETTINGS_FILE) then return nil end
		return HttpService:JSONDecode(readfile(SETTINGS_FILE))
	end)
	if not ok or type(data) ~= "table" then return nil end
	loadingSettings = true
	if type(data.sliders) == "table" then
		for _, s in ipairs(allSliders) do
			local v = data.sliders[s.key]
			if type(v) == "number" then s.set(v) end
		end
	end
	if type(data.chaos) == "number" then
		setChaos((data.chaos - CONFIG.chaosMin) / (CONFIG.chaosMax - CONFIG.chaosMin))
	end
	if type(data.toggles) == "table" then
		for _, t in ipairs(allToggles) do
			if not t.noSave and data.toggles[t.key] == true then t.set(true) end
		end
	end
	loadingSettings = false
	return data
end
